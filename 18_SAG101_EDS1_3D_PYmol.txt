The script below draws a dashed line between the actual closest heavy atoms in each model, so readers can see exactly what was measured. It also prints the P12 Lys NZ-to-Glu oxygen distance separately to check whether the model supports a putative salt bridge.

python
from math import dist
from pymol import cmd

def closest_atoms(selection_a, selection_b):
    atoms_a = cmd.get_model(f"({selection_a}) and not elem H").atom
    atoms_b = cmd.get_model(f"({selection_b}) and not elem H").atom
    if not atoms_a or not atoms_b:
        raise ValueError(f"Empty atom selection: {selection_a} or {selection_b}")
    a, b = min(
        ((a, b) for a in atoms_a for b in atoms_b),
        key=lambda pair: dist(pair[0].coord, pair[1].coord),
    )
    return a, b, dist(a.coord, b.coord)

for model, partner in (("p9", "asn59"), ("p12", "lys59")):
    a, b, distance = closest_atoms(f"{model} and glu257", partner)
    print(f"{model}: Glu257 {a.name} to {b.resn}{b.resi} {b.name}: {distance:.2f} A")
    cmd.distance(
        f"gap_{model}",
        f"{model} and index {a.index}",
        f"{model} and index {b.index}",
        cutoff=30.0,
        mode=0,
    )

a, b, distance = closest_atoms("p12 and glu257 and name OE1+OE2", "lys59 and name NZ")
print(f"P12 charged atoms: Glu257 {a.name} to Lys59 {b.name}: {distance:.2f} A")
python end

from pathlib import Path

source = Path("render_all_SAG101_interfaces.pml")

fetch 7xjp, pocket_ref, type=cif, async=0
align (pocket_ref and chain A and name CA), (pg001 and chain A and name CA)

color magenta, ligand_ref
deselect

"""

script = script.replace(marker, ligand_setup + marker, 1)
for group in ("001", "002", "003", "004"):
    script = script.replace(
        f"enable pg{group}\n",
        f"enable pg{group}\nenable pocket_ref\n",
    )

        f"SAG101_PG{group}_interface.png",
        f"SAG101_PG{group}_interface_ligand.png",
    )

    "SAG101_four_interfaces.pse",
    "SAG101_four_interfaces_ligand.pse",
)
