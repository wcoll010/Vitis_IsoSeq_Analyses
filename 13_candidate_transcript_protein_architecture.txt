#!/usr/bin/env python3

from pathlib import Path
import re


BASE = Path(
    "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"
)


OUTDIR = (
    BASE
    / "WGCNA_Vrip_X_Vrup"
    / "signed_network"
    / "candidate_architecture"
    / "final_multigenotype_targets"
)


ID_FILE = (
    OUTDIR
    / "Target_Genes_All_PB_Isoforms.txt"
)


PROTEIN_FASTA = (
    BASE
    / "eggnog_input"
    / "Vrip_X_Vrup_filtered_proteins.faa"
)


OUT_FASTA = (
    OUTDIR
    / "Target_Genes_All_Predicted_ORFs.faa"
)


ORF_MAP = (
    OUTDIR
    / "Target_Genes_ORF_Map.tsv"
)


target_ids = {
    x.strip()
    for x in ID_FILE.read_text().splitlines()
    if x.strip()
}


print(
    "Target PB isoforms:",
    len(target_ids)
)


def fasta_records(path):

    header = None
    seq = []

    with open(path) as fh:

        for line in fh:

            line = line.rstrip("\n")

            if line.startswith(">"):

                if header is not None:
                    yield header, "".join(seq)

                header = line[1:]
                seq = []

            else:

                seq.append(
                    line.strip()
                )

        if header is not None:

            yield header, "".join(seq)


records = []
found_isoforms = set()


with open(
    OUT_FASTA,
    "w"
) as fasta_out:

    for header, seq in fasta_records(
        PROTEIN_FASTA
    ):

        parts = re.split(
            r"\s+",
            header,
            maxsplit=1
        )


        if len(parts) < 2:
            continue


        isoform_id = parts[0]


        if isoform_id not in target_ids:
            continue


        orf_info = parts[1].split("|")


        if len(orf_info) < 5:
            continue


        protein_id = orf_info[0]

        protein_length = int(
            orf_info[1]
            .replace(
                "_aa",
                ""
            )
        )

        strand = orf_info[2]

        cds_start = int(
            float(
                orf_info[3]
            )
        )

        cds_end = int(
            float(
                orf_info[4]
            )
        )


        seq = seq.rstrip("*")


        fasta_out.write(
            f">{protein_id}\n"
        )

        fasta_out.write(
            f"{seq}\n"
        )


        records.append(
            (
                isoform_id,
                protein_id,
                protein_length,
                strand,
                cds_start,
                cds_end
            )
        )


        found_isoforms.add(
            isoform_id
        )


with open(
    ORF_MAP,
    "w"
) as out:

    out.write(
        "Isoform_ID\t"
        "Protein_ID\t"
        "Protein_Length_aa\t"
        "ORF_Strand\t"
        "CDS_Start_nt\t"
        "CDS_End_nt\n"
    )

    for row in records:

        out.write(
            "\t".join(
                map(
                    str,
                    row
                )
            )
            + "\n"
        )


missing = sorted(
    target_ids -
    found_isoforms
)


print(
    "Predicted ORF records:",
    len(records)
)


print(
    "Isoforms with >=1 predicted ORF:",
    len(found_isoforms)
)


print(
    "Isoforms without predicted ORF:",
    len(missing)
)


if missing:

    print(
        "\nIsoforms without predicted protein:"
    )

    for x in missing:

        print(
            " ",
            x
        )


print(
    "\nProtein FASTA:",
    OUT_FASTA
)


print(
    "ORF map:",
    ORF_MAP
)

read_fasta_simple <- function(file) {

    x <- readLines(file)

    headers <- which(
        startsWith(
            x,
            ">"
        )
    )

    out <- vector(
        "list",
        length(headers)
    )

    for (i in seq_along(headers)) {

        start <- headers[i]

        end <- if (
            i < length(headers)
        ) {

            headers[i + 1] - 1

        } else {

            length(x)
        }


        protein_id <- sub(
            "^>",
            "",
            x[start]
        )


        seq_lines <- if (
            end > start
        ) {

            x[
                (start + 1):end
            ]

        } else {

            character(0)
        }


        seq_string <- paste0(
            seq_lines,
            collapse = ""
        )


        seq_string <- gsub(
            "\\*",
            "",
            seq_string
        )


        out[[i]] <- data.frame(

            Protein_ID =
                protein_id,

            Protein_Sequence =
                seq_string,

            Sequence_Length =
                nchar(
                    seq_string
                ),

            stringsAsFactors =
                FALSE
        )
    }


    bind_rows(
        out
    )
}

make_letters <- function(n) {

    if (n <= 26) {

        return(
            LETTERS[
                seq_len(n)
            ]
        )
    }


    paste0(
        "P",
        seq_len(n)
    )
}


protein_groups <- representative_orf %>%

    left_join(

        master %>%

            dplyr::select(
                Gene_ID,
                Isoform_ID
            ) %>%

            distinct(),

        by = "Isoform_ID"
    ) %>%

    group_by(
        Gene_ID
    ) %>%

    group_modify(
        ~ {

            x <- .x


            valid_seq <- (
                !is.na(
                    x$Protein_Sequence
                ) &
                x$Protein_Sequence != ""
            )


            x$Protein_Group <- NA_character_


            if (any(valid_seq)) {

                unique_sequences <- unique(
                    x$Protein_Sequence[
                        valid_seq
                    ]
                )


                letters_here <- make_letters(
                    length(
                        unique_sequences
                    )
                )


                x$Protein_Group[
                    valid_seq
                ] <- paste(
                    "Protein",
                    letters_here[
                        match(
                            x$Protein_Sequence[
                                valid_seq
                            ],
                            unique_sequences
                        )
                    ]
                )
            }


            x
        }
    ) %>%

    ungroup()

# 9A. SHORT DOMAIN LABELS FOR FIGURES
# ============================================================

short_domain_label <- function(
    full_annotation,
    signature_description
) {

    x <- paste(
        full_annotation,
        signature_description
    )


    case_when(

        # NB-ARC nucleotide-binding domain
        str_detect(
            x,
            regex(
                "NB[- ]?ARC",
                ignore_case = TRUE
            )
        ) ~

            "NB-ARC",


        # Winged-helix domain
        str_detect(
            x,
            regex(
                "winged.?helix",
                ignore_case = TRUE
            )
        ) ~

            "WHD",


        # TIR domain
        str_detect(
            x,
            regex(
                "Toll/interleukin|TIR domain|TIR[-_ ]",
                ignore_case = TRUE
            )
        ) ~

            "TIR",


        # Leucine-rich repeat domains
        str_detect(
            x,
            regex(
                "leucine.?rich repeat|LRR",
                ignore_case = TRUE
            )
        ) ~

            "LRR",


        # WRKY DNA-binding domain
        str_detect(
            x,
            regex(
                "WRKY",
                ignore_case = TRUE
            )
        ) ~

            "WRKY",


        # EDS1 / EP domain
        str_detect(
            x,
            regex(
                "EDS1.*EP|EP domain",
                ignore_case = TRUE
            )
        ) ~

            "EDS1/EP",


        # Fungal-lipase-like domain
        str_detect(
            x,
            regex(
                "fungal lipase",
                ignore_case = TRUE
            )
        ) ~

            "Lipase-like",


        # Wall-associated kinase / galacturonan binding
        str_detect(
            x,
            regex(
                "wall.?associated.*galacturonan|WAK.*binding|GUB_WAK",
                ignore_case = TRUE
            )
        ) ~

            "WAK/GUB",


        # Protein kinase domain
        str_detect(
            x,
            regex(
                "protein kinase|Pkinase",
                ignore_case = TRUE
            )
        ) ~

            "Kinase",


        # RPW8
        str_detect(
            x,
            regex(
                "RPW8",
                ignore_case = TRUE
            )
        ) ~

            "RPW8",


        # Heavy-metal associated domain
        str_detect(
            x,
            regex(
                "heavy.?metal|HMA",
                ignore_case = TRUE
            )
        ) ~

            "HMA",


        # C-JID
        str_detect(
            x,
            regex(
                "C-JID",
                ignore_case = TRUE
            )
        ) ~

            "C-JID",


        # SBP
        str_detect(
            x,
            regex(
                "SBP",
                ignore_case = TRUE
            )
        ) ~

            "SBP",


        # Peptidase C48
        str_detect(
            x,
            regex(
                "Peptidase[_ -]?C48",
                ignore_case = TRUE
            )
        ) ~

            "Peptidase C48",


        # NAP
        str_detect(
            x,
            regex(
                "\\bNAP\\b",
                ignore_case = TRUE
            )
        ) ~

            "NAP",


        # If nothing above matches, retain the underlying
        # Pfam signature description rather than the longer
        # InterPro description.
        !is.na(
            signature_description
        ) &
            signature_description != "" &
            signature_description != "-" ~

            signature_description,


        TRUE ~

            full_annotation
    )
}

# 10. HELPERS
# ============================================================

safe_name <- function(x) {

    gsub(
        "[^A-Za-z0-9_.-]",
        "_",
        x
    )
}


protein_group_order <- function(x) {

    label <- sub(
        "^Protein ",
        "",
        x
    )


    number <- match(
        label,
        LETTERS
    )


    ifelse(
        is.na(number),
        999,
        number
    )
}


# ============================================================
# 11. MAKING FOUR-PANEL PLOTS FOR CANDIDATE GENES
#=============================================================

make_candidate_plot <- function(gene_id) {


    # --------------------------------------------------------
    # A. Gene data
    # --------------------------------------------------------

    dat_all <- master %>%

        filter(
            Gene_ID == gene_id
        ) %>%

        distinct(
            Isoform_ID,
            .keep_all = TRUE
        )


    if (nrow(dat_all) == 0) {

        stop(
            "No rows found for ",
            gene_id
        )
    }


    # --------------------------------------------------------
    # DISPLAY FILTER
    # --------------------------------------------------------

    dat <- dat_all %>%

        rowwise() %>%

        mutate(

            Max_Mean_IF =
                max(
                    c(
                        `101-14_Control`,
                        `101-14_Infested`,
                        `3309_Control`,
                        `3309_Infested`,
                        SW_Control,
                        SW_Infested
                    ),
                    na.rm = TRUE
                )
        ) %>%

        ungroup() %>%

        filter(
            is.finite(
                Max_Mean_IF
            ),
            Max_Mean_IF >=
                MIN_PLOT_IF
        )


    cat(
        gene_id,
        ": plotting ",
        nrow(dat),
        " of ",
        nrow(dat_all),
        " isoforms at IF >= ",
        MIN_PLOT_IF,
        "\n",
        sep = ""
    )


    if (nrow(dat) == 0) {

        warning(
            gene_id,
            " has no isoforms passing the plotting threshold."
        )

        return(NULL)
    }


    # --------------------------------------------------------
    # Candidate name
    # --------------------------------------------------------

    candidate_name <- unique(
        dat$Candidate
    )


    candidate_name <- candidate_name[
        !is.na(candidate_name) &
        candidate_name != ""
    ]


    if (length(candidate_name) == 0) {

        candidate_name <- gene_id

    } else {

        candidate_name <- candidate_name[1]
    }


    # --------------------------------------------------------
    # B. Order isoforms
    #
    # Same approach as previous cross-genotype figure: highest resistant infestation usage at top.
    # --------------------------------------------------------

    dat <- dat %>%

        mutate(

            Resistant_Infested_Mean =
                rowMeans(
                    cbind(
                        `3309_Infested`,
                        SW_Infested
                    ),
                    na.rm = TRUE
                ),


            Isoform_Label =
                paste0(

                    Isoform_ID,

                    ifelse(
                        Significant_Any_Contrast,
                        " *",
                        ""
                    ),

                    ifelse(
                        !is.na(PRGminer_Class) &
                            PRGminer_Class != "",

                        paste0(
                            " [",
                            PRGminer_Class,
                            "]"
                        ),

                        ""
                    )
                )
        ) %>%

        arrange(
            desc(
                Resistant_Infested_Mean
            ),
            Isoform_ID
        ) %>%

        mutate(

            y_pos =
                rev(
                    seq_len(
                        n()
                    )
                )
        )


    y_lookup <- dat %>%

        dplyr::select(
            Isoform_ID,
            y_pos
        )


    # ========================================================
    # PANEL 1 — ISOFORM USAGE
    # ========================================================

    heat <- dat %>%

        dplyr::select(
            Isoform_ID,
            y_pos,
            `101-14_Control`,
            `101-14_Infested`,
            `3309_Control`,
            `3309_Infested`,
            SW_Control,
            SW_Infested
        ) %>%

        pivot_longer(

            cols = c(
                `101-14_Control`,
                `101-14_Infested`,
                `3309_Control`,
                `3309_Infested`,
                SW_Control,
                SW_Infested
            ),

            names_to =
                "Group",

            values_to =
                "Isoform_Fraction"
        ) %>%

        mutate(

            Group =
                factor(

                    Group,

                    levels = c(
                        "101-14_Control",
                        "101-14_Infested",
                        "3309_Control",
                        "3309_Infested",
                        "SW_Control",
                        "SW_Infested"
                    ),

                    labels = c(
                        "101-14\nControl",
                        "101-14\nInfested",
                        "3309\nControl",
                        "3309\nInfested",
                        "SW\nControl",
                        "SW\nInfested"
                    )
                )
        )


    p_heat <- ggplot(
        heat,
        aes(
            x = Group,
            y = y_pos,
            fill = Isoform_Fraction
        )
    ) +

        geom_tile(
            width = 0.86,
            height = 0.72
        ) +

        scale_fill_gradient(
            low = "white",
            high = "steelblue",
            limits = c(
                0,
                1
            ),
            na.value = "grey95",
            name = "Mean isoform\nfraction"
        ) +

        scale_y_continuous(
            breaks =
                dat$y_pos,
            labels =
                dat$Isoform_Label,
            limits = c(
                0.5,
                nrow(dat) + 0.5
            )
        ) +

        labs(
            x = NULL,
            y = NULL,
            title = "Isoform usage"
        ) +

        guides(
            fill =
                guide_colorbar(
                    title.position = "top",
                    title.hjust = 0.5,
                    barwidth = 8,
                    barheight = 0.6
                )
        ) +

        theme_bw() +

        theme(

            panel.grid =
                element_blank(),

            axis.text.x =
                element_text(
                    angle = 45,
                    hjust = 1
                ),

            axis.text.y =
                element_text(
                    size = 8
                ),

            plot.title =
                element_text(
                    size = 10,
                    face = "bold"
                ),

            legend.position =
                "bottom",

            legend.direction =
                "horizontal",

            legend.box =
                "horizontal"
        )


    # ========================================================
    # PANEL 2 — TRANSCRIPT STRUCTURE
    # ========================================================

    exon_dat <- exons %>%

        filter(
            Isoform_ID %in%
                dat$Isoform_ID
        ) %>%

        inner_join(
            y_lookup,
            by = "Isoform_ID"
        )


    transcript_ranges <- exon_dat %>%

        group_by(
            Isoform_ID,
            y_pos
        ) %>%

        summarise(
            xmin = min(start),
            xmax = max(end),
            .groups = "drop"
        )


    p_rna <- ggplot() +

        geom_segment(
            data =
                transcript_ranges,
            aes(
                x = xmin,
                xend = xmax,
                y = y_pos,
                yend = y_pos
            ),
            linewidth = 0.5
        ) +

        geom_rect(
            data =
                exon_dat,
            aes(
                xmin = start,
                xmax = end,
                ymin = y_pos - 0.28,
                ymax = y_pos + 0.28
            )
        ) +

        scale_y_continuous(
            breaks =
                dat$y_pos,
            labels =
                rep(
                    "",
                    nrow(dat)
                ),
            limits = c(
                0.5,
                nrow(dat) + 0.5
            )
        ) +

        labs(
            x = "Genomic coordinate",
            y = NULL,
            title = "Transcript structure"
        ) +

        theme_bw() +

        theme(

            panel.grid =
                element_blank(),

            axis.text.y =
                element_blank(),

            axis.ticks.y =
                element_blank(),

            plot.title =
                element_text(
                    size = 10,
                    face = "bold"
                )
        )


    # ========================================================
    # PANEL 3 — PREDICTED PROTEIN ARCHITECTURE
    # ========================================================

    protein_ranges <- dat %>%

        filter(
            !is.na(
                Protein_ID
            ),
            Protein_ID != "",
            !is.na(
                Protein_Length_aa
            )
        ) %>%

        dplyr::select(
            Isoform_ID,
            y_pos,
            Protein_ID,
            Protein_Group,
            Protein_Length_aa,
            N_Predicted_ORFs
        ) %>%

        distinct() %>%

        mutate(

            Protein_Group_Label =
                ifelse(

                    !is.na(
                        N_Predicted_ORFs
                    ) &
                        N_Predicted_ORFs > 1,

                    paste0(
                        Protein_Group,
                        " (ORFs=",
                        N_Predicted_ORFs,
                        ")"
                    ),

                    Protein_Group
                )
        )


    pfam_gene <- protein_ranges %>%

        inner_join(
            pfam,
            by = "Protein_ID"
        )


    if (nrow(protein_ranges) > 0) {

        max_protein <- max(
            protein_ranges$Protein_Length_aa,
            na.rm = TRUE
        )

    } else {

        max_protein <- 1
    }


    p_protein <- ggplot() +

        # protein backbone
        geom_segment(
            data =
                protein_ranges,
            aes(
                x = 1,
                xend =
                    Protein_Length_aa,
                y =
                    y_pos,
                yend =
                    y_pos
            ),
            linewidth = 1
        ) +

        # Pfam domains
        geom_rect(
            data =
                pfam_gene,
            aes(
                xmin =
                    Domain_Start,
                xmax =
                    Domain_End,
                ymin =
                    y_pos - 0.25,
                ymax =
                    y_pos + 0.25,
                fill =
                    Domain_Label
            )
        ) +

        # Protein Group labels
        geom_text(
            data =
                protein_ranges,
            aes(
                x =
                    Protein_Length_aa +
                    max_protein * 0.025,
                y =
                    y_pos,
                label =
                    Protein_Group_Label
            ),
            hjust = 0,
            size = 3
        ) +

        scale_y_continuous(
            breaks =
                dat$y_pos,
            labels =

                    "",
                    nrow(dat)
                ),
            limits = c(
                0.5,
                nrow(dat) + 0.5
            )
        ) +

        coord_cartesian(
            xlim = c(
                0,
                max_protein * 1.30
            ),
            clip = "off"
        ) +

        labs(
            x = "Protein position (aa)",
            y = NULL,
            fill = "Pfam domain",
            title = "Predicted protein architecture"
        ) +

        guides(
            fill =
                guide_legend(
                    nrow = 2,
                    byrow = TRUE,
                    title.position = "top"
                )
        ) +

        theme_bw() +

        theme(

            panel.grid =
                element_blank(),

            axis.text.y =
                element_blank(),

            axis.ticks.y =
                element_blank(),

            plot.title =
                element_text(
                    size = 10,
                    face = "bold"
                ),

            legend.position =
                "bottom",

            legend.direction =
                "horizontal",

            legend.box =
                "horizontal",

            legend.text =
                element_text(
                    size = 8
                ),

            legend.title =
                element_text(
                    size = 9
                ),

            plot.margin =
                margin(
                    5,
                    10,
                    5,
                    5
                )
        )


    # ========================================================
    # PANEL 4 — PROTEIN GROUP ΔIF
    #
    # Same calculation as before: summed isoform fraction for transcripts encoding identical predicted proteins
    # ΔIF = Infested - Control; calculated separately for each genotype (101-14, 3309, SW)
    # ========================================================

    delta_base <- dat %>%

        filter(
            !is.na(
                Protein_Group
            ),
            Protein_Group != ""
        ) %>%

        group_by(
            Protein_Group
        ) %>%

        summarise(

            `101-14_Control` =
                sum(
                    `101-14_Control`,
                    na.rm = TRUE
                ),

            `101-14_Infested` =
                sum(
                    `101-14_Infested`,
                    na.rm = TRUE
                ),

            `3309_Control` =
                sum(
                    `3309_Control`,
                    na.rm = TRUE
                ),

            `3309_Infested` =
                sum(
                    `3309_Infested`,
                    na.rm = TRUE
                ),

            SW_Control =
                sum(
                    SW_Control,
                    na.rm = TRUE
                ),

            SW_Infested =
                sum(
                    SW_Infested,
                    na.rm = TRUE
                ),

            .groups = "drop"
        ) %>%

        mutate(

            `101-14` =
                `101-14_Infested` -
                `101-14_Control`,

            `3309` =
                `3309_Infested` -
                `3309_Control`,

            SW =
                SW_Infested -
                SW_Control,

            Protein_Group_Order =
                protein_group_order(
                    Protein_Group
                )
        ) %>%

        arrange(
            Protein_Group_Order
        )


    delta_dat <- delta_base %>%

        dplyr::select(

            `101-14`,
            `3309`,
            SW

        pivot_longer(

                `3309`,

    delta_dat$Genotype <- factor(
        delta_dat$Genotype,
        levels = c(

            "3309",
            "SW"
        )
    )


    delta_max <- max(

            delta_dat$Delta_IF

        na.rm = TRUE
    )


    if (
        !is.finite(
            delta_max
        ) ||
            delta_max == 0

        delta_max <- 0.05
    }


    delta_limit <-
        delta_max * 1.15


    p_delta <- ggplot(
        delta_dat,
        aes(
            x = Protein_Group,
            y = Delta_IF,
            fill = Genotype
        )

            yintercept = 0,
            linetype = "dashed",
            linewidth = 0.45
        ) +

        geom_col(
            position =
                position_dodge(
                    width = 0.8
                ),
            width = 0.72
        ) +

        coord_cartesian(
            ylim = c(
                -delta_limit,
                delta_limit
            )
        ) +

        labs(
            x =
                "Predicted protein group",
            y =
                "Change in summed isoform fraction\n(Infested − Control)",
            title =
                "Predicted protein-group change",
            fill =
                "Genotype"
        ) +

        guides(
            fill =
                guide_legend(
                    nrow = 1,
                    title.position = "top"
                )
        ) +

        theme_bw() +

        theme(

            panel.grid =
                element_blank(),

            axis.text.x =
                element_text(
                    angle = 45,
                    hjust = 1,
                    vjust = 1,
                    size = 8
                ),

            axis.text.y =
                element_text(
                    size = 8
                ),

            axis.title.x =
                element_text(
                    size = 9
                ),

            axis.title.y =
                element_text(
                    size = 9
                ),

            plot.title =
                element_text(
                    size = 10,
                    face = "bold"
                ),

            legend.position =
                "bottom",

            legend.direction =
                "horizontal",

            legend.box =
                "horizontal"
        )


    # ========================================================
    # SAVE DISPLAYED ISOFORM TABLE
    # ========================================================

    displayed_isoforms <- dat %>%

        dplyr::select(
            Gene_ID,
            Candidate,
            Isoform_ID,
            Significant_Any_Contrast,
            N_Significant_Contrasts,
            Significant_Contrasts,
            PRGminer_Class,
            `101-14_Control`,
            `101-14_Infested`,
            `3309_Control`,
            `3309_Infested`,
            SW_Control,
            SW_Infested,
            Max_Mean_IF,
            Protein_ID,
            Protein_Group,
            Protein_Length_aa,
            N_Predicted_ORFs
        )


    write.csv(
        displayed_isoforms,
        file.path(
            PLOT_DIR,
            paste0(
                safe_name(
                    gene_id
                ),
                "_displayed_isoforms_IFge0.10.csv"
            )
        ),
        row.names = FALSE
    )


    # ========================================================
    # COMBINE
    # ========================================================

    n_iso <- nrow(
        dat
    )


    fig_height <- max(
        5,
        1.8 +
            n_iso * 0.55
    )


    title_text <- paste0(
        candidate_name,
        "\n",
        gene_id
    )


    combined <- (

        p_heat |

            p_rna |

            p_protein |

            p_delta

    ) +

        plot_layout(

            widths = c(
                1.55,
                2.6,
                3.2,
                2.4
            )
        ) +

        plot_annotation(

            title =
                title_text,

            subtitle =
                paste0(
                    "* significant in ≥1 of the 9 final DIU contrasts; ",
                    "only isoforms reaching mean IF ≥ ",
                    MIN_PLOT_IF,
                    " in ≥1 genotype × treatment group are shown; ",
                    "identical predicted amino-acid sequences share a Protein Group"
                )
        )


    list(
        plot = combined,
        height = fig_height,
        delta = delta_dat,
        displayed_isoforms = displayed_isoforms
    )
}
