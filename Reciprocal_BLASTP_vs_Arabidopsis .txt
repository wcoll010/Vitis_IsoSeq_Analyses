#!/usr/bin/env python3

import re
from pathlib import Path

BASE = Path(
    "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2/"
    "reciprocal_BLAST_Arabidopsis"
)

INFILE = (
    BASE /
    "vitis/Vrip_X_Vrup_reference_transcript_proteins.faa"
)

OUT_ALL = (
    BASE /
    "vitis/Vrip_X_Vrup_longest_protein_per_gene.faa"
)

OUT_VRP = (
    BASE /
    "vitis/VrpB38_longest_protein_per_gene.faa"
)

OUT_VRI = (
    BASE /
    "vitis/Vri588271_longest_protein_per_gene.faa"
)

OUT_MAP = (
    BASE /
    "vitis/Vrip_X_Vrup_gene_representatives.tsv"
)


def read_fasta(path):

    header = None
    seq = []

    with open(path) as f:

        for line in f:

            line = line.rstrip()

            if not line:
                continue

            if line.startswith(">"):

                if header is not None:
                    yield header, "".join(seq)

                header = line[1:]
                seq = []

            else:
                seq.append(line)

        if header is not None:
            yield header, "".join(seq)


best = {}

for header, seq in read_fasta(INFILE):

    transcript = header.split()[0]

    # e.g.
    # ...g265620.t02 -> ...g265620
    gene = re.sub(r"\.t\d+$", "", transcript)

    if gene.startswith("VITVrpB38_"):
        parent = "VrpB38"

    elif gene.startswith("VITVri588271_"):
        parent = "Vri588271"

    else:
        parent = "other"

    current = best.get(gene)

    if (
        current is None
        or len(seq) > len(current["seq"])
        or (
            len(seq) == len(current["seq"])
            and transcript < current["transcript"]
        )
    ):
        best[gene] = {
            "gene": gene,
            "transcript": transcript,
            "parent": parent,
            "seq": seq
        }


print("Total genes:", len(best))

print(
    "VrpB38 genes:",
    sum(x["parent"] == "VrpB38" for x in best.values())
)

print(
    "Vri588271 genes:",
    sum(x["parent"] == "Vri588271" for x in best.values())
)


def write_fasta(records, path):

    with open(path, "w") as out:

        for x in records:

            out.write(
                f">{x['gene']} "
                f"transcript={x['transcript']} "
                f"length={len(x['seq'])}\n"
            )

            seq = x["seq"]

            for i in range(0, len(seq), 60):
                out.write(seq[i:i+60] + "\n")


records = [
    best[g]
    for g in sorted(best)
]

write_fasta(
    records,
    OUT_ALL
)

write_fasta(
    [x for x in records if x["parent"] == "VrpB38"],
    OUT_VRP
)

write_fasta(
    [x for x in records if x["parent"] == "Vri588271"],
    OUT_VRI
)


with open(OUT_MAP, "w") as out:

    out.write(
        "Gene_ID\tRepresentative_Transcript\t"
        "Parent\tProtein_Length_aa\n"
    )

    for x in records:

        out.write(
            f"{x['gene']}\t"
            f"{x['transcript']}\t"
            f"{x['parent']}\t"
            f"{len(x['seq'])}\n"
        )


print("\nWritten:")
print(OUT_ALL)
print(OUT_VRP)
print(OUT_VRI)
print(OUT_MAP)

python3 "$RBH/scripts/make_parent_gene_proteomes.py"

OUTDIR="$RBH/SAG101_g229560_g229570_check"

mkdir -p "$OUTDIR"

module load ncbi-blast/2.13.0+

# ============================================================
# IDs
# ============================================================

G1="VITVrpB38_v1.0.hap1.chr14.ver1.0.g229560"
G2="VITVrpB38_v1.0.hap1.chr14.ver1.0.g229570"

# 1. Extract the two reference proteins
# ============================================================

    -p "$T1" \
    "$REF_PROT" \
    > "$OUTDIR/g229560.faa"

seqkit grep \
    -p "$T2" \
    "$REF_PROT" \
    > "$OUTDIR/g229570.faa"

seqkit fx2tab -n -l \
    "$OUTDIR/g229560.faa" \
    "$OUTDIR/g229570.faa"

echo


# ============================================================
# 2. Pairwise BLASTP
#
# g229570 is query so qcovs tells us how much of the short
# protein can be aligned to g229560.
# ============================================================

blastp \
    -query "$OUTDIR/g229570.faa" \
    -subject "$OUTDIR/g229560.faa" \
    -evalue 1e-5 \
    -max_hsps 10 \
    -outfmt "6 qseqid sseqid pident length qlen slen qstart qend sstart send evalue bitscore" \
    > "$OUTDIR/g229570_vs_g229560_blastp.tsv"


echo "============================================================"
echo "g229570 protein vs g229560 protein"
echo "============================================================"

column -t \
    "$OUTDIR/g229570_vs_g229560_blastp.tsv"

echo

# 3. Print reference annotation for both genes
# ============================================================

echo "============================================================"

echo "============================================================"

grep -E "${G1}|${G2}" "$REF_GTF" \
    > "$OUTDIR/g229560_g229570_reference_GTF.tsv"

grep -E "${G1}|${G2}" "$REF_GTF"

#!/usr/bin/env python3

import re
from collections import defaultdict
from pathlib import Path

BASE = Path(
    "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"
)

RBH = BASE / "reciprocal_BLAST_Arabidopsis"

ISO_GTF = (
    BASE /
    "sqanti_results/Vrip_X_Vrup_sqanti/filter_results/"
    "Vrip_X_Vrup_filtered.filtered.gtf"
)

OUTDIR = RBH / "SAG101_g229560_g229570_check"
OUTDIR.mkdir(parents=True, exist_ok=True)

OUT = OUTDIR / "IsoSeq_exon_bridge_classification.tsv"

CHR = "VITVrpB38_v1.0.hap1.chr14"

# Reference gene boundaries
G1_START = 30977955
G1_END   = 30984486

G2_START = 30985426
G2_END   = 30986837


def overlaps(a1, a2, b1, b2):
    return a1 <= b2 and a2 >= b1


# ------------------------------------------------------------
# Collect exon coordinates for every transcript in this region
# ------------------------------------------------------------

tx_exons = defaultdict(list)
tx_strand = {}

with open(ISO_GTF) as f:

    for line in f:

        if line.startswith("#"):
            continue

        fields = line.rstrip().split("\t")

        if len(fields) != 9:
            continue

        chrom = fields[0]
        feature = fields[2]

        if chrom != CHR or feature != "exon":
            continue

        start = int(fields[3])
        end = int(fields[4])

        # Ignore exons completely outside the local region
        if end < G1_START - 5000:
            continue

        if start > G2_END + 20000:
            continue

        m = re.search(
            r'transcript_id "([^"]+)"',
            fields[8]
        )

        if not m:
            continue

        tx = m.group(1)

        tx_exons[tx].append(
            (start, end)
        )

        tx_strand[tx] = fields[6]


# ------------------------------------------------------------
# Classify each transcript
# ------------------------------------------------------------

rows = []

for tx, exons in tx_exons.items():

    exons = sorted(exons)

    hit_g1 = []
    hit_g2 = []

    for i, (start, end) in enumerate(exons, start=1):

        if overlaps(
            start, end,
            G1_START, G1_END
        ):
            hit_g1.append(i)

        if overlaps(
            start, end,
            G2_START, G2_END
        ):
            hit_g2.append(i)

    overlaps_g1 = len(hit_g1) > 0
    overlaps_g2 = len(hit_g2) > 0

    if overlaps_g1 and overlaps_g2:
        classification = "BRIDGES_BOTH"

    elif overlaps_g1:
        classification = "G229560_ONLY"

    elif overlaps_g2:
        classification = "G229570_ONLY"

    else:
        classification = "NEITHER"

    exon_string = ";".join(
        f"{s}-{e}"
        for s, e in exons
    )

    rows.append([
        tx,
        tx_strand.get(tx, ""),
        min(s for s, e in exons),
        max(e for s, e in exons),
        len(exons),
        overlaps_g1,
        overlaps_g2,
        ",".join(map(str, hit_g1)) if hit_g1 else "",
        ",".join(map(str, hit_g2)) if hit_g2 else "",
        classification,
        exon_string
    ])


rows.sort(
    key=lambda x: (x[2], x[3], x[0])
)


with open(OUT, "w") as out:

    out.write(
        "Transcript_ID\t"
        "Strand\t"
        "Start\t"
        "End\t"
        "Exon_Count\t"
        "Overlaps_g229560\t"
        "Overlaps_g229570\t"
        "g229560_Exon_Numbers\t"
        "g229570_Exon_Numbers\t"
        "Classification\t"
        "Exon_Coordinates\n"
    )

    for row in rows:
        out.write(
            "\t".join(map(str, row)) +
            "\n"
        )


# ------------------------------------------------------------
# Console summary
# ------------------------------------------------------------

counts = defaultdict(int)

for row in rows:
    counts[row[9]] += 1


print("==============================================")
print("TRANSCRIPT CLASSIFICATION")
print("==============================================")

for key in [
    "BRIDGES_BOTH",
    "G229560_ONLY",
    "G229570_ONLY",
    "NEITHER"
]:
    print(
        f"{key}: {counts[key]}"
    )


print()
print("==============================================")
print("TRANSCRIPTS WITH EXONS IN BOTH GENE MODELS")
print("==============================================")

for row in rows:

    if row[9] == "BRIDGES_BOTH":

        print(
            row[0],
            f"exons={row[4]}",
            f"g229560_exons={row[7]}",
            f"g229570_exons={row[8]}",
            f"coords={row[10]}",
            sep="\t"
        )


print()
print("Output:")
print(OUT)

python3 "$RBH/scripts/check_SAG101_exon_bridging.py"
