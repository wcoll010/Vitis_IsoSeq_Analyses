### run some QC checks to make sure the new reference genomes are correct


GENOME_DIR="/rhome/wcoll010/shared/pacbio/genomes"
HYBRIDS=("Vber_X_Vrup" "Vrip_X_Vber" "Vrip_X_Vrup" "Vrip_X_Vace")

for HYB in "${HYBRIDS[@]}"; do
    echo "=================================================="
    echo "QC REPORT FOR: $HYB"
    echo "=================================================="
    
    FASTA="${GENOME_DIR}/${HYB}/${HYB}_genome.fasta"
    GFF="${GENOME_DIR}/${HYB}/${HYB}.gff3"
    
    # 1. Total chromosomes in FASTA
    FASTA_SEQS=$(grep -c "^>" "$FASTA")
    echo "[1] Total FASTA sequences (Expected: 38): $FASTA_SEQS"
    
    # 2. Check for accidental hap2 presence in FASTA headers
    HAP2_FASTA=$(grep "^>" "$FASTA" | grep -ci "hap2" || true)
    echo "[2] Hap2 headers in FASTA (Expected: 0): $HAP2_FASTA"
    
    # 3. Check for accidental hap2 presence in GFF3
    HAP2_GFF=$(grep -v "^#" "$GFF" | awk '{print $1}' | grep -ci "hap2" || true)
    echo "[3] Hap2 features in GFF3 (Expected: 0): $HAP2_GFF"
    
    # 4. Check FASTA vs. GFF3 chromosome header congruence
    grep "^>" "$FASTA" | tr -d ">" | awk '{print $1}' | sort -u > /tmp/${HYB}_fasta_ids.txt
    grep -v "^#" "$GFF" | awk '{print $1}' | sort -u > /tmp/${HYB}_gff_ids.txt
    DIFF_COUNT=$(comm -3 /tmp/${HYB}_fasta_ids.txt /tmp/${HYB}_gff_ids.txt | wc -l)
    echo "[4] FASTA vs GFF3 ID mismatches (Expected: 0): $DIFF_COUNT"
    
    # 5. Total annotated gene count
    GENES=$(awk '$3 == "gene"' "$GFF" | wc -l)
    echo "[5] Total annotated genes (Expected: ~60,000-66,000): $GENES"
    echo ""
done

#!/bin/bash -l
#SBATCH --job-name=pbmm2_p2
#SBATCH --cpus-per-task=12
#SBATCH --mem=100G
#SBATCH --time=48:00:00
#SBATCH --array=1-32%8
#SBATCH -p epyc
#SBATCH --output=logs/align_p2_%A_%a.out
#SBATCH --error=logs/align_p2_%A_%a.err
#SBATCH --mail-user=wcoll010@ucr.edu
#SBATCH --mail-type=ALL

set -euo pipefail

MANIFEST="/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2/manifest_part2.tsv"
REF_BASE="/rhome/wcoll010/shared/pacbio/genomes"
OUT_BASE="/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2/aligned_diploid_samples"

source /opt/linux/rhel/8.x/x86_64/pkgs/miniconda3/py39_4.12.0/etc/profile.d/conda.sh
conda activate "$HOME/.conda/envs/isoseq_run_env"

# 1. Parse sample information from manifest
LINE=$(sed -n "${SLURM_ARRAY_TASK_ID}p" "$MANIFEST")
SAMPLE=$(echo "$LINE" | cut -f1)
INPUT_BAM=$(echo "$LINE" | cut -f2)
GENOME_FOLDER=$(echo "$LINE" | cut -f3)

# 2. Set sample output directory and target BAM
SAMPLE_OUT_DIR="${OUT_BASE}/${SAMPLE}"
mkdir -p "$SAMPLE_OUT_DIR"
OUT_BAM="${SAMPLE_OUT_DIR}/${SAMPLE}_mapped.bam"

# 3. Locate the .mmi index file
REF=$(ls ${REF_BASE}/${GENOME_FOLDER}/*.mmi | head -n 1)

echo "=== Processing Sample: $SAMPLE (Task ${SLURM_ARRAY_TASK_ID}) ==="
echo "Input BAM:  $INPUT_BAM"
echo "Reference:  $REF"
echo "Output BAM: $OUT_BAM"

if [[ ! -f "$INPUT_BAM" ]]; then
    echo "ERROR: Input file $INPUT_BAM does not exist!"
    exit 1
fi

# 4. Run pbmm2 alignment
/usr/bin/time -v pbmm2 align "$REF" "$INPUT_BAM" "$OUT_BAM" \
    --preset ISOSEQ \
    --sort \
    --num-threads "${SLURM_CPUS_PER_TASK}" \
    --log-level INFO

echo "Alignment complete for $SAMPLE -> $OUT_BAM"

### Generate summary file with alignment statistics

#!/bin/bash -l
#SBATCH --job-name=align_qc
#SBATCH --cpus-per-task=2
#SBATCH --mem=32G
#SBATCH --time=02:00:00
#SBATCH --output=logs/align_qc_%j.out
#SBATCH --error=logs/align_qc_%j.err
#SBATCH --mail-user=wcoll010@ucr.edu
#SBATCH --mail-type=ALL

set -euo pipefail

# 1. Environment Setup
source /opt/linux/rhel/8.x/x86_64/pkgs/miniconda3/py39_4.12.0/etc/profile.d/conda.sh
conda activate "$HOME/.conda/envs/isoseq_run_env"

BASE_DIR="/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"
MANIFEST="${BASE_DIR}/manifest_final.csv"
BAM_DIR="${BASE_DIR}/aligned_diploid_samples"
OUT_CSV="${BASE_DIR}/alignment_qc_summary.csv"

mkdir -p "${BASE_DIR}/logs"

echo "=================================================="
echo "Generating Alignment QC Summary"
echo "Manifest: $MANIFEST"
echo "Output:   $OUT_CSV"
echo "=================================================="

# 2. Write CSV Header
echo "sample_id,reference_genome,total_reads,mapped_reads,pct_mapped,unmapped_reads,mapq_0_reads,pct_mapq_0,mapq_1_29_reads,mapq_ge30_reads,pct_mapq_ge30" > "$OUT_CSV"

# 3. Process Each Sample
while IFS=$'\t' read -r SAMPLE FLNC_PATH GENOME; do
    # Skip empty lines or headers if present
    [[ -z "$SAMPLE" || "$SAMPLE" == "Genotype" || "$SAMPLE" == "sample_id" ]] && continue
    
    BAM_PATH="${BAM_DIR}/${SAMPLE}/${SAMPLE}_mapped.bam"
    
    if [[ ! -f "$BAM_PATH" ]]; then
        echo "WARNING: ${BAM_PATH} not found. Recording as missing."
        echo "${SAMPLE},${GENOME},NA,NA,NA,NA,NA,NA,NA,NA,NA" >> "$OUT_CSV"
        continue
    fi
    
    echo "Processing $SAMPLE..."
    
    # Extract comprehensive mapping and MAPQ metrics in a single streaming pass
    STATS=$(samtools view -@ 1 "$BAM_PATH" | awk '
    BEGIN {
        total = 0;
        mapped = 0;
        unmapped = 0;
        q0 = 0;
        q_low = 0;
        q_high = 0;
    }
    {
        total++;
        if ($3 == "*") {
            unmapped++;
        } else {
            mapped++;
            if ($5 == 0) {
                q0++;
            } else if ($5 < 30) {
                q_low++;
            } else {
                q_high++;
            }
        }
    }
    END {
        pct_map   = (total > 0)  ? (mapped / total * 100) : 0;
        pct_q0    = (mapped > 0) ? (q0 / mapped * 100)    : 0;
        pct_qhigh = (mapped > 0) ? (q_high / mapped * 100): 0;
        printf "%d,%d,%.2f,%d,%d,%.2f,%d,%d,%.2f", total, mapped, pct_map, unmapped, q0, pct_q0, q_low, q_high, pct_qhigh;
    }')
    
    echo "${SAMPLE},${GENOME},${STATS}" >> "$OUT_CSV"

done < "$MANIFEST"

echo "=================================================="
echo "QC Complete. Results written to: $OUT_CSV"
echo "=================================================="

### collapse aligned data to make a master trascriptome for each hybrid reference transcriptome/catalog
### perform via isoseq collapse command


#!/bin/bash -l
#SBATCH --job-name=pool_collapse
#SBATCH --cpus-per-task=8
#SBATCH --mem=80G
#SBATCH --time=12:00:00
#SBATCH --array=1-6
#SBATCH -p epyc
#SBATCH --output=logs/pool_collapse_%A_%a.out
#SBATCH --error=logs/pool_collapse_%A_%a.err
#SBATCH --mail-user=wcoll010@ucr.edu
#SBATCH --mail-type=ALL

set -euo pipefail

# 1. Environment Setup
source /opt/linux/rhel/8.x/x86_64/pkgs/miniconda3/py39_4.12.0/etc/profile.d/conda.sh
conda activate "$HOME/.conda/envs/isoseq_run_env"

BASE_DIR="/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"
IN_DIR="${BASE_DIR}/aligned_diploid_samples"
OUT_DIR="${BASE_DIR}/master_transcriptomes"
mkdir -p "$OUT_DIR" logs

# 2. Define the 6 Backgrounds and Matching Genotype Prefixes
GROUPS=("Vrip_X_Vrup" "Vber_X_Vrup" "Vrip_X_Vber" "Vrip_X_Vace" "CabSauv08" "Vripa")
GENOS=("sw 3309 101-14" "99R 1103" "5BB SO4" "1616" "cab33" "ripa")

# Map SLURM_ARRAY_TASK_ID (1-6) to 0-indexed bash arrays
INDEX=$((SLURM_ARRAY_TASK_ID - 1))
GROUP="${GROUPS[$INDEX]}"
GENO_LIST="${GENOS[$INDEX]}"

echo "=========================================="
echo "Task ID: $SLURM_ARRAY_TASK_ID"
echo "Processing Group: $GROUP"
echo "Target Genotypes: $GENO_LIST"
echo "=========================================="

# 3. Gather Mapped BAMs for this Specific Group
BAMS=()
for GENO in $GENO_LIST; do
    for BAM in ${IN_DIR}/${GENO}_*/*_mapped.bam; do
        if [[ -f "$BAM" ]]; then
            BAMS+=("$BAM")
        fi
    done
done

echo "Found ${#BAMS[@]} BAM files for $GROUP."

if [[ ${#BAMS[@]} -eq 0 ]]; then
    echo "ERROR: No BAM files found matching prefixes: $GENO_LIST"
    exit 1
fi

POOLED_BAM="${OUT_DIR}/${GROUP}_pooled.sorted.bam"

# 4. Merge and Sort
echo "Merging and sorting ${#BAMS[@]} BAMs for $GROUP..."
samtools merge -@ 4 - "${BAMS[@]}" | \
samtools sort -@ 4 -m 12G -o "$POOLED_BAM"
samtools index -@ 4 "$POOLED_BAM"

# 5. Collapse Redundant Isoforms
echo "Running isoseq collapse for $GROUP..."
isoseq collapse --do-not-collapse-extra-5exons \
    --num-threads "${SLURM_CPUS_PER_TASK}" \
    "$POOLED_BAM" \
    "${OUT_DIR}/${GROUP}_master.gff"

echo "Successfully built Master Transcriptome for $GROUP!"
