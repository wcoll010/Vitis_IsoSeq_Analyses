# ============================================================
# 4. FASTA READER
# ============================================================

def read_fasta(path):

    records = {}

    header = None
    seq = []


    with open(path) as fh:

        for line in fh:

            line = line.strip()

            if not line:
                continue


            if line.startswith(">"):

                if header is not None:

                    records[header] = "".join(
                        seq
                    )


                header = line[1:]

                seq = []


            else:

                seq.append(
                    line
                )


        if header is not None:

            records[header] = "".join(
                seq
            )


    return records

# ============================================================
# 6. SELECT EXACT PROTEINS
# ============================================================

def get_unique_by_pattern(
    records,
    pattern
):

    matches = []


    for header, seq in records.items():

        # Headers use either:
        #
        #   field__field__field
        #
        # or:
        #
        #   field|field|field
        #
        # Split into exact header fields so that, for example,
        # Protein_P1 does NOT accidentally match Protein_P17.

        tokens = re.split(
            r"__|\|",
            header
        )


        if pattern in tokens:

            matches.append(
                (
                    header,
                    seq
                )
            )


    if len(matches) == 0:

        raise ValueError(
            f"No FASTA record matched exact label: {pattern}"
        )


    unique_sequences = {}

    for header, seq in matches:

        unique_sequences.setdefault(
            seq,
            []
        ).append(
            header
        )


    if len(unique_sequences) != 1:

        print()
        print(
            f"ERROR: {pattern} matched multiple unique sequences:"
        )

        for sequence, headers in unique_sequences.items():

            print(
                f"  sequence length = {len(sequence)} aa"
            )

            for header in headers:

                print(
                    f"    {header}"
                )

        raise ValueError(
            f"Exact label {pattern} matched "
            f"{len(unique_sequences)} unique sequences"
        )


    sequence = next(
        iter(
            unique_sequences
        )
    )


    headers = unique_sequences[
        sequence
    ]


    return sequence, headers

# ============================================================

VALID_AA = set(

)


for name, sequence in proteins.items():

    sequence = (
        sequence
        .replace("*", "")
        .upper()
    )

    proteins[name] = sequence


    bad = set(
        sequence
    ) - VALID_AA


    if bad:

        raise ValueError(
            f"{name} contains unexpected characters: "
            f"{sorted(bad)}"
        )


    observed = len(
        sequence
    )

    expected = EXPECTED_LENGTHS[
        name
    ]


    if observed != expected:

        raise ValueError(
            f"{name}: expected {expected} aa, "
            f"observed {observed} aa"
        )

#!/usr/bin/env python3

from pathlib import Path
import json
import csv
import re
import statistics


# ============================================================
# 1. PATHS
# ============================================================

BASE = Path(
    "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"
)

AFDIR = (
    BASE
    / "WGCNA_Vrip_X_Vrup"
    / "signed_network"
    / "candidate_architecture"
    / "AlphaFold3_EDS1_SAG101_PAD4"
)

OUTPUTS = AFDIR / "outputs"
ANALYSIS = AFDIR / "analysis"
TABLES = ANALYSIS / "tables"
STRUCTURES = ANALYSIS / "structures"

TABLES.mkdir(parents=True, exist_ok=True)
STRUCTURES.mkdir(parents=True, exist_ok=True)


# ============================================================
# 2. FIND INDIVIDUAL SEED/SAMPLE RESULTS
# ============================================================

pattern = re.compile(
    r"seed-(\d+)_sample-(\d+)"
)

rows = []


for summary_file in sorted(
    OUTPUTS.glob(
        "*/*/seed-*_sample-*/*_summary_confidences.json"
    )
):

    sample_dir = summary_file.parent

    match = pattern.search(sample_dir.name)

    if match is None:
        continue

    seed = int(match.group(1))
    sample = int(match.group(2))

    # Complex is two levels above seed directory
    complex_name = sample_dir.parent.name

    with open(summary_file) as fh:
        data = json.load(fh)

    chain_iptm = data.get("chain_iptm", [None, None])
    chain_ptm = data.get("chain_ptm", [None, None])

    chain_pair_iptm = data.get(
        "chain_pair_iptm",
        [[None, None], [None, None]]
    )

    chain_pair_pae_min = data.get(
        "chain_pair_pae_min",
        [[None, None], [None, None]]
    )

    model_file = next(
        sample_dir.glob("*_model.cif"),
        None
    )

    row = {
        "complex": complex_name,
        "seed": seed,
        "sample": sample,

        "ranking_score": data.get(
            "ranking_score"
        ),

        "iptm": data.get("iptm"),
        "ptm": data.get("ptm"),

        "fraction_disordered": data.get(
            "fraction_disordered"
        ),

        "has_clash": data.get(
            "has_clash"
        ),

        "chain_A_iptm": (
            chain_iptm[0]
            if len(chain_iptm) > 0
            else None
        ),

        "chain_B_iptm": (
            chain_iptm[1]
            if len(chain_iptm) > 1
            else None
        ),

        "chain_A_ptm": (
            chain_ptm[0]
            if len(chain_ptm) > 0
            else None
        ),

        "chain_B_ptm": (
            chain_ptm[1]
            if len(chain_ptm) > 1
            else None
        ),

        "A_to_B_pair_iptm": (
            chain_pair_iptm[0][1]
        ),

        "B_to_A_pair_iptm": (
            chain_pair_iptm[1][0]
        ),

        "A_to_B_pae_min": (
            chain_pair_pae_min[0][1]
        ),

        "B_to_A_pae_min": (
            chain_pair_pae_min[1][0]
        ),

        "summary_file": str(
            summary_file
        ),

        "model_file": (
            str(model_file)
            if model_file is not None
            else ""
        )
    }

    rows.append(row)

#!/usr/bin/env python3

from pathlib import Path
import json
import csv
import statistics

BASE = Path(
    "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"
)

AFDIR = (
    BASE
    / "WGCNA_Vrip_X_Vrup"
    / "signed_network"
    / "candidate_architecture"
    / "AlphaFold3_EDS1_SAG101_PAD4"
)

OUTPUTS = AFDIR / "outputs"
TABLES = AFDIR / "analysis" / "tables"

TABLES.mkdir(parents=True, exist_ok=True)

rows = []

for complex_dir in sorted(OUTPUTS.iterdir()):

    if not complex_dir.is_dir():
        continue

    complex_name = complex_dir.name

    inner = complex_dir / complex_name

    conf = inner / f"{complex_name}_confidences.json"
    summary = inner / f"{complex_name}_summary_confidences.json"

    if not conf.exists() or not summary.exists():
        continue

    with open(conf) as fh:
        x = json.load(fh)

    with open(summary) as fh:
        s = json.load(fh)

    pae = x["pae"]
    chain_ids = x["token_chain_ids"]
    res_ids = x["token_res_ids"]

    chains = sorted(set(chain_ids))

    if len(chains) != 2:
        print(
            f"Skipping {complex_name}: "
            f"expected 2 chains, found {chains}"
        )
        continue

    chain_a = chains[0]
    chain_b = chains[1]

    idx_a = [
        i for i, c in enumerate(chain_ids)
        if c == chain_a
    ]

    idx_b = [
        i for i, c in enumerate(chain_ids)
        if c == chain_b
    ]

    ab = [
        pae[i][j]
        for i in idx_a
        for j in idx_b
    ]

    ba = [
        pae[i][j]
        for i in idx_b
        for j in idx_a
    ]

    combined = ab + ba

    def frac_below(values, cutoff):
        return sum(v < cutoff for v in values) / len(values)

    rows.append({
        "complex": complex_name,

        "chain_A":
            chain_a,

        "chain_B":
            chain_b,

        "n_res_A":
            len(idx_a),

        "n_res_B":
            len(idx_b),

        "iptm":
            s.get("iptm"),

        "ptm":
            s.get("ptm"),

        "ranking_score":
            s.get("ranking_score"),

        "mean_interchain_PAE":
            statistics.mean(combined),

        "median_interchain_PAE":
            statistics.median(combined),

        "min_interchain_PAE":
            min(combined),

        "fraction_interchain_PAE_lt5":
            frac_below(combined, 5),

        "fraction_interchain_PAE_lt10":
            frac_below(combined, 10),

        "fraction_interchain_PAE_lt15":
            frac_below(combined, 15)
    })


outfile = TABLES / "AlphaFold3_interface_PAE_summary.csv"

fields = list(rows[0].keys())

with open(outfile, "w", newline="") as out:

    writer = csv.DictWriter(
        out,
        fieldnames=fields
    )

    writer.writeheader()
    writer.writerows(rows)


print()
print("=" * 72)
print("ALPHAFOLD 3 INTERFACE PAE SUMMARY")
print("=" * 72)
print()

for row in rows:

    print(row["complex"])

    print(
        f"  ipTM:                    "
        f"{row['iptm']:.3f}"
    )

    print(
        f"  mean interchain PAE:     "
        f"{row['mean_interchain_PAE']:.2f} Å"
    )

    print(
        f"  median interchain PAE:   "
        f"{row['median_interchain_PAE']:.2f} Å"
    )

    print(
        f"  minimum interchain PAE:  "
        f"{row['min_interchain_PAE']:.2f} Å"

    print(

print("Output:")

]:

# ============================================================
# 3. FASTA READER
# ============================================================

def read_fasta(
    path
):

    records = {}

    header = None
    sequence = []


    with open(
        path
    ) as fh:

        for line in fh:

            line = line.strip()


            if not line:

                continue


            if line.startswith(
                ">"
            ):

                if header is not None:

                    records[
                        header
                    ] = "".join(
                        sequence
                    )


                header = line[
                    1:
                ]


                sequence = []


            else:

                sequence.append(
                    line
                )


        if header is not None:

            records[
                header
            ] = "".join(
                sequence
            )


    return records

# Same exact-field matching used previously.
# ============================================================

def get_unique_by_pattern(
    records,
    pattern
):

    matches = []


    for header, sequence in records.items():

        tokens = re.split(
            r"__|\|",
            header
        )


        if pattern in tokens:

            matches.append(
                (
                    header,
                    sequence
                )
            )


    if len(
        matches
    ) == 0:

        raise ValueError(
            f"No FASTA record matched exact label: "
            f"{pattern}"
        )


    unique_sequences = {}


    for header, sequence in matches:

        sequence = (
            sequence
            .replace(
                "*",
                ""
            )
            .upper()
        )


        unique_sequences.setdefault(
            sequence,
            []
        ).append(
            header
        )


    if len(
        unique_sequences
    ) != 1:

        raise ValueError(
            f"{pattern} matched "
            f"{len(unique_sequences)} unique sequences"
        )


    sequence = next(
        iter(
            unique_sequences
        )
    )


    return (
        sequence,
        unique_sequences[
            sequence
        ]
    )

def numeric_or_none(
    x
):

    if x is None:

    try:

# ============================================================
# 6. HELPER:
# LOCATE FULL CONFIDENCE FILE FOR A SUMMARY FILE
#
# AF3 normally gives:
#
#   *_summary_confidences.json
#   *_confidences.json
#
# in the same directory.
# ============================================================

def confidence_for_summary(
    summary_file
):

    name = summary_file.name


    if not name.endswith(
        "_summary_confidences.json"
    ):

        return None


    confidence_name = name.replace(
        "_summary_confidences.json",
        "_confidences.json"
    )


    confidence_file = (
        summary_file.parent
        / confidence_name
    )


    if confidence_file.exists():

        return confidence_file


    # --------------------------------------------------------
    # Fallback:
    # if exact paired name is absent, accept a single
    # confidence JSON in the same directory.
    # --------------------------------------------------------

    candidates = [

        x

        for x in summary_file.parent.glob(
            "*_confidences.json"
        )

        if "_summary_confidences.json"
        not in x.name
    ]


    if len(
        candidates
    ) == 1:

        return candidates[
            0
        ]


    return None

def seed_sample_from_path(

    seed = None

    for part in path.parts:

            "seed-"

                left, right = part.split(
                    "_sample-"
                )


                seed = int(
                    left.replace(

                        ""
                    )
                )


                sample = int(
                    right
                )


            except:

                pass


    return (
        seed,
        sample

# 12. INTER-CHAIN PAE FUNCTION
# ============================================================

def analyze_interface(
    complex_name,
    model
):

    confidence_file = Path(
        model[
            "confidence_file"
        ]
    )


    with open(
        confidence_file
    ) as fh:

        confidence = json.load(
            fh
        )


    if "pae" not in confidence:

        raise ValueError(
            f"{complex_name}: confidence JSON lacks 'pae'\n"
            f"{confidence_file}"
        )


    if "token_chain_ids" not in confidence:

        raise ValueError(
            f"{complex_name}: confidence JSON lacks "
            f"'token_chain_ids'\n"
            f"{confidence_file}"
        )


    pae = np.asarray(
        confidence[
            "pae"
        ],
        dtype=float
    )


    chain_ids = list(
        confidence[
            "token_chain_ids"
        ]
    )


    unique_chains = []


    for chain in chain_ids:

        if chain not in unique_chains:

            unique_chains.append(
                chain
            )


    if len(
        unique_chains
    ) != 2:

        raise ValueError(
            f"{complex_name}: expected 2 chains, "
            f"found {unique_chains}"
        )


    chain_a = unique_chains[
        0
    ]


    chain_b = unique_chains[
        1
    ]


    idx_a = np.array(

        [
            i

            for i, chain in enumerate(
                chain_ids
            )

            if chain == chain_a
        ],

        dtype=int
    )


    idx_b = np.array(

        [
            i

            for i, chain in enumerate(
                chain_ids
            )

            if chain == chain_b
        ],

        dtype=int
    )


    if (
        len(
            idx_a
        ) == 0
        or
        len(
            idx_b
        ) == 0
    ):

        raise ValueError(
            f"{complex_name}: empty chain index."
        )


    # --------------------------------------------------------
    # Directional blocks:
    #
    #   A -> B
    #   B -> A
    #
    # As in the previous analysis, combine both directions.
    # --------------------------------------------------------

    ab = pae[
        np.ix_(
            idx_a,
            idx_b
        )
    ].ravel()


    ba = pae[
        np.ix_(
            idx_b,
            idx_a
        )
    ].ravel()


    combined = np.concatenate(
        [
            ab,
            ba
        ]
    )


    def fraction_below(
        cutoff
    ):

        return float(
            np.mean(
                combined < cutoff
            )
        )


    left, right = complex_name.split(
        "__"
    )


    eds1 = left.replace(
        "EDS1_",
        ""
    )


    sag101 = right.replace(
        "SAG101_",
        ""
    )


    return {

        "complex":
            complex_name,

        "EDS1":
            eds1,

        "SAG101":
            sag101,

        "best_seed":
            model[
                "seed"
            ],

        "best_sample":
            model[
                "sample"
            ],

        "ranking_score":
            model[
                "ranking_score"
            ],

        "iptm":
            model[
                "iptm"
            ],

        "ptm":
            model[
                "ptm"
            ],

        "chain_A":
            chain_a,

        "chain_B":
            chain_b,

        "n_tokens_A":
            len(
                idx_a
            ),

        "n_tokens_B":
            len(
                idx_b
            ),

        "n_interchain_values":
            len(
                combined
            ),

        "mean_interchain_PAE":
            float(
                np.mean(
                    combined
                )
            ),

        "median_interchain_PAE":
            float(
                np.median(
                    combined
                )
            ),

        "min_interchain_PAE":
            float(
                np.min(
                    combined
                )
            ),

        "fraction_interchain_PAE_lt5":
            fraction_below(
                5
            ),

        "fraction_interchain_PAE_lt10":
            fraction_below(
                10
            ),

        "fraction_interchain_PAE_lt15":
            fraction_below(
                15
            ),

        "summary_file":
            model[
                "summary_file"
            ],

        "confidence_file":
            model[
                "confidence_file"
            ]
    }

def plot_heatmap(
    matrix,
    title,
    xlabel,
    ylabel,
    cbar_label,
    outfile,
    decimals=2,
    vmin=None,
    vmax=None
):

    data = matrix.to_numpy(
        dtype=float
    )


    n_rows = len(
        matrix.index
    )


    n_cols = len(
        matrix.columns
    )


    width = max(
        4.8,
        1.4 * n_cols +
        2.5
    )


    height = max(
        4.2,
        0.9 * n_rows +
        2.0
    )


    fig, ax = plt.subplots(
        figsize=(
            width,
            height
        )
    )


    im = ax.imshow(

        data,

        aspect="auto",

        vmin=vmin,

        vmax=vmax
    )


    cbar = fig.colorbar(
        im,
        ax=ax,
        shrink=0.82
    )


    cbar.set_label(
        cbar_label
    )


    ax.set_xticks(
        np.arange(
            n_cols
        )
    )


    ax.set_xticklabels(
        matrix.columns
    )


    ax.set_yticks(
        np.arange(
            n_rows
        )
    )


    ax.set_yticklabels(
        matrix.index
    )


    ax.set_xlabel(
        xlabel
    )


    ax.set_ylabel(
        ylabel
    )


    ax.set_title(
        title
    )


    # --------------------------------------------------------
    # Cell values
    # --------------------------------------------------------

    for i in range(
        n_rows
    ):

        for j in range(
            n_cols
        ):

            value = data[
                i,
                j
            ]


            if np.isnan(
                value
            ):

                label = "NA"

            else:

                label = (
                    f"{value:.{decimals}f}"
                )


            ax.text(

                j,
                i,
                label,

                ha="center",
                va="center",

                fontsize=10
            )


    fig.tight_layout()


    fig.savefig(

        outfile.with_suffix(
            ".pdf"
        ),

        bbox_inches="tight"
    )


    fig.savefig(

        outfile.with_suffix(
            ".png"
        ),

        dpi=600,

        bbox_inches="tight"
    )


    plt.close(
        fig
    )

    try:

        x = float(
            value
        )


        if math.isnan(
            x
        ):

            return None


        return x


    except:

        return None


# ============================================================
# 5. FIND BEST MODEL FOR A COMPLEX
# ============================================================

def find_best_model(
    complex_name
):

    complex_dir = (
        OUTPUTS
        / complex_name
    )


    if not complex_dir.exists():

        print(
            "Missing complex directory:",
            complex_dir
        )

        return None


    summary_files = sorted(

        complex_dir.rglob(
            "*_summary_confidences.json"
        )
    )


    candidates = []


    for summary_file in summary_files:

        try:

            with open(
                summary_file
            ) as fh:

                summary = json.load(
                    fh
                )


        except Exception as error:

            print(
                "Could not read:",
                summary_file
            )

            print(
                error
            )

            continue


        # ----------------------------------------------------
        # Find corresponding full confidence file
        # ----------------------------------------------------

        confidence_name = (
            summary_file.name.replace(
                "_summary_confidences.json",
                "_confidences.json"
            )
        )


        confidence_file = (
            summary_file.parent
            / confidence_name
        )


        # ----------------------------------------------------
        # Fallback if exact paired filename not present
        # ----------------------------------------------------

        if not confidence_file.exists():

            fallback = [

                x

                for x in summary_file.parent.glob(
                    "*_confidences.json"
                )

                if "_summary_confidences.json"
                not in x.name
            ]


            if len(
                fallback
            ) == 1:

                confidence_file = fallback[
                    0
                ]


        if not confidence_file.exists():

            continue


        candidates.append({

            "summary_file":
                summary_file,

            "confidence_file":
                confidence_file,

            "ranking_score":
                safe_float(
                    summary.get(
                        "ranking_score"
                    )
                ),

            "iptm":
                safe_float(
                    summary.get(
                        "iptm"
                    )
                ),

            "ptm":
                safe_float(
                    summary.get(
                        "ptm"
                    )
                )
        })


    if len(
        candidates
    ) == 0:

        print(
            "No usable models found for:",
            complex_name
        )

        return None


    def score(
        row
    ):

        ranking = (
            row[
                "ranking_score"
            ]
            if row[
                "ranking_score"
            ] is not None
            else -np.inf
        )


        iptm = (
            row[
                "iptm"
            ]
            if row[
                "iptm"
            ] is not None
            else -np.inf
        )


        ptm = (
            row[
                "ptm"
            ]
            if row[
                "ptm"
            ] is not None
            else -np.inf
        )


        return (
            ranking,
            iptm,
            ptm
        )


    return max(
        candidates,
        key=score
    )
