### In-depth analysis of the SAG101 locus in V. riparia

### Step 1: extract and analyze the RNA isforms of the g309600/VriSAG101 locus


# run R script 
#!/usr/bin/env Rscript

.libPaths(c(
    "/rhome/wcoll010/bigdata_groenlab/R_libs",
    .libPaths()
))

suppressPackageStartupMessages({
    library(dplyr)
    library(tidyr)
    library(stringr)
    library(ggplot2)
})


# ============================================================
# 1. SETTINGS
# ============================================================

BASE <- "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"

TARGET_GENE <-
    "VITVri588271_v1.0.hap1.chr14.ver1.0.g309600"


OUTDIR <- file.path(
    BASE,
    "WGCNA_Vrip_X_Vrup",
    "signed_network",
    "candidate_architecture",
    "g309600_locus_validation"
)


dir.create(
    OUTDIR,
    recursive = TRUE,
    showWarnings = FALSE
)


# ============================================================
# 2. INPUT FILES
# ============================================================

SQANTI_FILE <- file.path(
    BASE,
    "sqanti_results",
    "Vrip_X_Vrup_sqanti",
    "filter_results",
    "Vrip_X_Vrup_filtered_RulesFilter_result_classification.txt"
)


GTF_FILE <- file.path(
    BASE,
    "sqanti_results",
    "Vrip_X_Vrup_sqanti",
    "filter_results",
    "Vrip_X_Vrup_filtered.filtered.gtf"
)


RAW_COUNTS <- file.path(
    BASE,
    "oarfish_matrices",
    "Vrip_X_Vrup_oarfish_counts.tsv"
)


FINAL_MASTER <- file.path(
    BASE,
    "WGCNA_Vrip_X_Vrup",
    "signed_network",
    "candidate_architecture",
    "final_multigenotype_targets",
    "Target_Genes_All_Isoforms_CrossGenotype_Master.csv"
)


EXON_MASTER <- file.path(
    BASE,
    "WGCNA_Vrip_X_Vrup",
    "signed_network",
    "candidate_architecture",
    "final_multigenotype_targets",
    "Target_Genes_All_Isoforms_Exons.csv"
)


required <- c(
    SQANTI_FILE,
    GTF_FILE,
    RAW_COUNTS,
    FINAL_MASTER
)


missing <- required[
    !file.exists(required)
]


if (length(missing) > 0) {

    stop(
        "Missing required file(s):\n",
        paste(
            missing,
            collapse = "\n"
        )
    )
}


# ============================================================
# 3. READ SQANTI
# ============================================================

sqanti <- read.delim(
    SQANTI_FILE,
    header = TRUE,
    sep = "\t",
    quote = "",
    stringsAsFactors = FALSE,
    check.names = FALSE
)


if (
    !all(
        c(
            "isoform",
            "associated_gene"
        ) %in%
        colnames(sqanti)
    )
) {

    stop(
        "SQANTI file must contain isoform and associated_gene."
    )
}


gene_sqanti <- sqanti %>%

    filter(
        associated_gene == TARGET_GENE
    ) %>%

    mutate(
        Isoform_ID =
            as.character(
                isoform
            )
    )


cat("\n")
cat("====================================================\n")
cat("g309600 SQANTI TRANSCRIPTS\n")
cat("====================================================\n\n")


cat(
    "Total SQANTI isoforms assigned to g309600:",
    nrow(gene_sqanti),
    "\n"
)


cat(
    "Unique PB isoforms:",
    n_distinct(
        gene_sqanti$Isoform_ID
    ),
    "\n\n"
)


# Save all SQANTI annotation columns for this gene

write.csv(
    gene_sqanti,
    file.path(
        OUTDIR,
        "g309600_all_SQANTI_isoforms.csv"
    ),
    row.names = FALSE
)


# ============================================================
# 4. READ FILTERED ISO-SEQ GTF
# ============================================================

gtf <- read.delim(
    GTF_FILE,
    header = FALSE,
    sep = "\t",
    quote = "",
    comment.char = "#",
    stringsAsFactors = FALSE
)


colnames(gtf) <- c(
    "seqname",
    "source",
    "feature",
    "start",
    "end",
    "score",
    "strand",
    "frame",
    "attribute"
)


gtf$Isoform_ID <- str_match(
    gtf$attribute,
    'transcript_id "([^"]+)"'
)[, 2]


target_ids <- unique(
    gene_sqanti$Isoform_ID
)


gene_exons <- gtf %>%

    filter(
        feature == "exon",
        Isoform_ID %in%
            target_ids
    ) %>%

    dplyr::select(
        Isoform_ID,
        seqname,
        strand,
        start,
        end
    ) %>%

    arrange(
        Isoform_ID,
        start
    )


if (nrow(gene_exons) == 0) {

    stop(
        "No exon rows found for g309600 transcripts."
    )
}


write.csv(
    gene_exons,
    file.path(
        OUTDIR,
        "g309600_all_isoform_exons.csv"
    ),
    row.names = FALSE
)


# ============================================================
# 5. TRANSCRIPT GENOMIC RANGES
# ============================================================

ranges <- gene_exons %>%

    group_by(
        Isoform_ID
    ) %>%

    summarise(

        Chromosome =
            first(
                seqname
            ),

        Strand =
            first(
                strand
            ),

        Transcript_Start =
            min(
                start
            ),

        Transcript_End =
            max(
                end
            ),

        Genomic_Span_bp =
            Transcript_End -
            Transcript_Start +
            1,

        Exon_Count =
            n(),

        Exonic_Length_bp =
            sum(
                end -
                start +
                1
            ),

        .groups =
            "drop"
    )


write.csv(
    ranges,
    file.path(
        OUTDIR,
        "g309600_isoform_genomic_ranges.csv"
    ),
    row.names = FALSE
)


cat(
    "\nTranscript genomic ranges:\n"
)


print(
    ranges %>%
        arrange(
            Transcript_Start
        )
)


# ============================================================
# 6. JOIN FINAL DISPLAY / PROTEIN INFORMATION
#
# This lets us identify the four focal transcripts:
#
# PB.18476.92
# PB.18476.113
# PB.18476.73
# PB.18476.6
#
# and their predicted Protein Groups.
# ============================================================

master <- read.csv(
    FINAL_MASTER,
    stringsAsFactors = FALSE,
    check.names = FALSE
)


gene_master <- master %>%

    filter(
        Gene_ID == TARGET_GENE
    ) %>%

    distinct(
        Isoform_ID,
        .keep_all = TRUE
    )


ranges_annotated <- ranges %>%

    left_join(
        gene_master %>%

            dplyr::select(
                Isoform_ID,
                Candidate,
                Significant_Any_Contrast,
                N_Significant_Contrasts,
                Significant_Contrasts,
                PRGminer_Class,
                `101-14_Control`,
                `101-14_Infested`,
                `3309_Control`,
                `3309_Infested`,
                SW_Control,
                SW_Infested
            ),

        by = "Isoform_ID"
    )


write.csv(
    ranges_annotated,
    file.path(
        OUTDIR,
        "g309600_isoform_ranges_with_usage.csv"
    ),
    row.names = FALSE
)


# ============================================================
# 7. IDENTIFY GENOMIC CLUSTERS AUTOMATICALLY
#
# The suspicious pattern is that one transcript architecture
# is several kb away from the others.
#
# We identify transcript clusters by genomic overlap.
# ============================================================

ranges_ordered <- ranges %>%

    arrange(
        Transcript_Start,
        Transcript_End
    )


cluster_id <- integer(
    nrow(
        ranges_ordered
    )
)


if (nrow(ranges_ordered) > 0) {

    cluster <- 1L

    cluster_id[1] <- cluster

    current_end <-
        ranges_ordered$Transcript_End[1]


    if (nrow(ranges_ordered) > 1) {

        for (
            i in 2:nrow(ranges_ordered)
        ) {

            if (
                ranges_ordered$Transcript_Start[i] >
                    current_end
            ) {

                cluster <-
                    cluster + 1L
            }


            cluster_id[i] <-
                cluster


            current_end <-
                max(
                    current_end,
                    ranges_ordered$Transcript_End[i]
                )
        }
    }
}


ranges_ordered$Genomic_Cluster <-
    paste0(
        "Cluster_",
        cluster_id
    )


cluster_summary <- ranges_ordered %>%

    group_by(
        Genomic_Cluster
    ) %>%

    summarise(

        Cluster_Start =
            min(
                Transcript_Start
            ),

        Cluster_End =
            max(
                Transcript_End
            ),

        Cluster_Span_bp =
            Cluster_End -
            Cluster_Start +
            1,

        N_Transcripts =
            n(),

        Isoforms =
            paste(
                Isoform_ID,
                collapse = " | "
            ),

        .groups =
            "drop"
    )


cat("\n")
cat("====================================================\n")
cat("GENOMIC TRANSCRIPT CLUSTERS\n")
cat("====================================================\n\n")


print(
    cluster_summary
)


write.csv(
    cluster_summary,
    file.path(
        OUTDIR,
        "g309600_genomic_cluster_summary.csv"
    ),
    row.names = FALSE
)


# ============================================================
# 8. TEST WHETHER ANY TRANSCRIPT BRIDGES CLUSTERS
# ============================================================

if (nrow(cluster_summary) >= 2) {

    cluster_summary <- cluster_summary %>%

        arrange(
            Cluster_Start
        )


    upstream <- cluster_summary[
        1,
    ]


    downstream <- cluster_summary[
        nrow(cluster_summary),
    ]


    UP_START <-
        upstream$Cluster_Start

    UP_END <-
        upstream$Cluster_End

    DOWN_START <-
        downstream$Cluster_Start

    DOWN_END <-
        downstream$Cluster_End


    overlaps <- function(
        a1,
        a2,
        b1,
        b2
    ) {

        a1 <= b2 &
            a2 >= b1
    }


    bridge_classification <- gene_exons %>%

        group_by(
            Isoform_ID
        ) %>%

        summarise(

            Has_Upstream_Exon =
                any(
                    overlaps(
                        start,
                        end,
                        UP_START,
                        UP_END
                    )
                ),

            Has_Downstream_Exon =
                any(
                    overlaps(
                        start,
                        end,
                        DOWN_START,
                        DOWN_END
                    )
                ),

            N_Upstream_Exons =
                sum(
                    overlaps(
                        start,
                        end,
                        UP_START,
                        UP_END
                    )
                ),

            N_Downstream_Exons =
                sum(
                    overlaps(
                        start,
                        end,
                        DOWN_START,
                        DOWN_END
                    )
                ),

            .groups =
                "drop"
        ) %>%

        mutate(

            Bridges_Clusters =
                Has_Upstream_Exon &
                Has_Downstream_Exon
        ) %>%

        left_join(
            ranges,
            by = "Isoform_ID"
        ) %>%

        arrange(
            desc(
                Bridges_Clusters
            ),
            desc(
                Genomic_Span_bp
            )
        )


    cat("\n")
    cat("====================================================\n")
    cat("BRIDGING TRANSCRIPT TEST\n")
    cat("====================================================\n\n")


    cat(
        "Upstream region:",
        UP_START,
        "-",
        UP_END,
        "\n"
    )


    cat(
        "Downstream region:",
        DOWN_START,
        "-",
        DOWN_END,
        "\n\n"
    )


    print(
        bridge_classification
    )


    write.csv(
        bridge_classification,
        file.path(
            OUTDIR,
            "g309600_bridge_classification.csv"
        ),
        row.names = FALSE
    )


    bridging_ids <- bridge_classification %>%

        filter(
            Bridges_Clusters
        ) %>%

        pull(
            Isoform_ID
        )


    cat(
        "\nNumber of true exon-bridging Iso-Seq transcripts:",
        length(
            bridging_ids
        ),
        "\n"
    )


    if (
        length(
            bridging_ids
        ) > 0
    ) {

        cat(
            "\nBridging transcripts:\n"
        )

        print(
            bridging_ids
        )

    } else {

        cat(
            "\nNo transcript contains exons in both extreme regions.\n"
        )
    }

} else {

    warning(
        "Only one overlapping transcript cluster was detected. ",
        "No upstream/downstream bridge test was necessary."
    )

    bridge_classification <-
        data.frame()

    bridging_ids <-
        character(0)
}


# ============================================================
# 9. RAW OARFISH COUNTS
#
# Determine whether any bridge transcript is actually observed
# in 101-14, 3309, and/or SW.
# ============================================================

counts <- read.delim(
    RAW_COUNTS,
    header = TRUE,
    sep = "\t",
    quote = "",
    stringsAsFactors = FALSE,
    check.names = FALSE
)


if (
    !"isoform" %in%
    colnames(counts)
) {

    stop(
        "Raw count matrix is missing 'isoform' column."
    )
}


sample_cols <- setdiff(
    colnames(counts),
    "isoform"
)


gene_counts <- counts %>%

    rename(
        Isoform_ID =
            isoform
    ) %>%

    filter(
        Isoform_ID %in%
            target_ids
    ) %>%

    pivot_longer(

        cols =
            all_of(
                sample_cols
            ),

        names_to =
            "Sample",

        values_to =
            "Count"
    ) %>%

    mutate(

        Count =
            as.numeric(
                Count
            ),

        Genotype =
            case_when(

                grepl(
                    "^101-14_",
                    Sample
                ) ~
                    "101-14",

                grepl(
                    "^3309_",
                    Sample
                ) ~
                    "3309",

                grepl(
                    "^sw_",
                    Sample,
                    ignore.case = TRUE
                ) ~
                    "SW",

                TRUE ~
                    NA_character_
            ),

        Treatment =
            case_when(

                grepl(
                    "_con_",
                    Sample,
                    ignore.case = TRUE
                ) ~
                    "Control",

                grepl(
                    "_inf_",
                    Sample,
                    ignore.case = TRUE
                ) ~
                    "Infested",

                TRUE ~
                    NA_character_
            )
    ) %>%

    filter(
        !is.na(
            Genotype
        ),
        !is.na(
            Treatment
        )
    )


# ============================================================
# 10. SAMPLE-LEVEL SUPPORT SUMMARY
# ============================================================

support_summary <- gene_counts %>%

    group_by(
        Isoform_ID,
        Genotype,
        Treatment
    ) %>%

    summarise(

        Mean_Count =
            mean(
                Count,
                na.rm = TRUE
            ),

        Median_Count =
            median(
                Count,
                na.rm = TRUE
            ),

        Max_Count =
            max(
                Count,
                na.rm = TRUE
            ),

        N_Samples =
            n(),

        N_Samples_With_Count_GT0 =
            sum(
                Count > 0,
                na.rm = TRUE
            ),

        .groups =
            "drop"
    )


write.csv(
    support_summary,
    file.path(
        OUTDIR,
        "g309600_isoform_raw_count_support.csv"
    ),
    row.names = FALSE
)

# ============================================================
# 13. EXON ARCHITECTURE PLOT
# ============================================================

plot_exons <- gene_exons %>%

    inner_join(
        ranges_annotated %>%

            dplyr::select(
                Isoform_ID,
                `101-14_Control`,
                `101-14_Infested`,
                `3309_Control`,
                `3309_Infested`,
                SW_Control,
                SW_Infested
            ),

        by = "Isoform_ID"
    ) %>%

    mutate(

        Max_Mean_IF =
            pmax(
                `101-14_Control`,
                `101-14_Infested`,
                `3309_Control`,
                `3309_Infested`,
                SW_Control,
                SW_Infested,
                na.rm = TRUE
            )
    )


# show transcripts with >=1% mean usage anywhere
# plus every bridging transcript regardless of abundance

plot_ids <- plot_exons %>%

    filter(
        Max_Mean_IF >= 0.01 |
            Isoform_ID %in%
                bridging_ids
    ) %>%

    distinct(
        Isoform_ID
    ) %>%

    pull(
        Isoform_ID
    )


plot_dat <- gene_exons %>%

    filter(
        Isoform_ID %in%
            plot_ids
    )


plot_ranges <- plot_dat %>%

    group_by(
        Isoform_ID
    ) %>%

    summarise(

        xmin =
            min(
                start
            ),

        xmax =
            max(
                end
            ),

        .groups =
            "drop"
    )


iso_order <- plot_ranges %>%

    arrange(
        xmin
    ) %>%

    pull(
        Isoform_ID
    )


plot_dat$Isoform_ID <- factor(
    plot_dat$Isoform_ID,
    levels =
        rev(
            iso_order
        )
)


plot_ranges$Isoform_ID <- factor(
    plot_ranges$Isoform_ID,
    levels =
        rev(
            iso_order
        )
)


p <- ggplot() +

    geom_segment(
        data =
            plot_ranges,
        aes(
            x =
                xmin,
            xend =
                xmax,
            y =
                Isoform_ID,
            yend =
                Isoform_ID
        ),
        linewidth =
            0.45
    ) +

    geom_rect(
        data =
            plot_dat,
        aes(
            xmin =
                start,
            xmax =
                end,
            ymin =

                    Isoform_ID
                ) -
                0.28,

                as.numeric(
                    Isoform_ID
                ) +
                0.28
        )
    ) +

    labs(
        title =
            paste0(
                "Iso-Seq transcript architecture at ",
                TARGET_GENE
            ),
        x =
            "Genomic coordinate",
        y =
            "Isoform"
    ) +

    theme_bw() +

    theme(
        panel.grid =
            element_blank()
    )

read_fasta <- function(
    fasta_file
) {

    lines <- readLines(
        fasta_file
    )

    headers <- grep(
        "^>",
        lines
    )

    result <- vector(
        "list",
        length(headers)
    )


    for (
        i in seq_along(headers)
    ) {

        start_i <-
            headers[i]

        end_i <-
            if (
                i < length(headers)
            ) {
                headers[i + 1] - 1
            } else {
                length(lines)
            }


        protein_id <- sub(
            "^>",
            "",
            strsplit(
                lines[start_i],
                "\\s+"
            )[[1]][1]
        )


        sequence <- paste0(
            lines[
                (start_i + 1):end_i
            ],
            collapse = ""
        )


        result[[i]] <- data.frame(
            Protein_ID =
                protein_id,

            Protein_Sequence =
                sequence,

            stringsAsFactors =
                FALSE
        )
    }


    bind_rows(
        result
    )
}

def normalize_sample_name(x):

    y = x.lower()

    y = y.replace(
        ".",
        "_"
    )

    y = y.replace(
        "-",
        "_"
    )

    return y


def identify_genotype(sample):

    s = normalize_sample_name(
        sample
    )


    if (
        "101_14" in s
        or "10114" in s
    ):

        return "101-14"


    if "3309" in s:

        return "3309"


    if re.search(
        r"(^|_)sw(_|$)",
        s
    ):

        return "SW"


    if "so4" in s:

        return "SO4"


    if "5bb" in s:

        return "5BB"


    if "1616" in s:

        return "1616"


    if (
        "ripa" in s
        or "vripa" in s
    ):

        return "Ripa"


    return None


def identify_treatment(sample):

    s = normalize_sample_name(
        sample
    )


    if (
        "_inf_" in s
        or s.endswith(
            "_inf"
        )
        or "infested" in s
    ):

        return "Infested"


    if (
        "_con_" in s
        or s.endswith(
            "_con"
        )
        or "control" in s
        or "_ctrl_" in s
    ):

        return "Control"


    return None

CODON_TABLE = {

    "TTT":"F","TTC":"F","TTA":"L","TTG":"L",
    "TCT":"S","TCC":"S","TCA":"S","TCG":"S",
    "TAT":"Y","TAC":"Y","TAA":"*","TAG":"*",
    "TGT":"C","TGC":"C","TGA":"*","TGG":"W",

    "CTT":"L","CTC":"L","CTA":"L","CTG":"L",
    "CCT":"P","CCC":"P","CCA":"P","CCG":"P",
    "CAT":"H","CAC":"H","CAA":"Q","CAG":"Q",

    "ATT":"I","ATC":"I","ATA":"I","ATG":"M",
    "ACT":"T","ACC":"T","ACA":"T","ACG":"T",
    "AAT":"N","AAC":"N","AAA":"K","AAG":"K",
    "AGT":"S","AGC":"S","AGA":"R","AGG":"R",

    "GTT":"V","GTC":"V","GTA":"V","GTG":"V",
    "GCT":"A","GCC":"A","GCA":"A","GCG":"A",
    "GAT":"D","GAC":"D","GAA":"E","GAG":"E",
    "GGT":"G","GGC":"G","GGA":"G","GGG":"G"
}


def translate_orf(seq):

    if seq is None:
        return ""


    seq = (
        seq.upper()
        .replace(
            "U",
            "T"
        )
    )


    if (
        seq == ""
        or seq == "NA"
        or seq == "NAN"
    ):

        return ""


    aa = []


    for i in range(
        0,
        len(seq) - 2,
        3
    ):

        codon = seq[
            i:i + 3
        ]


        aa.append(
            CODON_TABLE.get(
                codon,
                "X"
            )
        )


    protein = "".join(
        aa
    )


    # Remove terminal stop for grouping.
    protein = protein.rstrip(
        "*"
    )


    return protein

def read_counts(path):

    with open(
        path
    ) as fh:

        reader = csv.reader(
            fh,
            delimiter="\t"
        )

        header = next(
            reader
        )


        samples = header[
            1:
        ]


        values = {}


        for row in reader:

            if not row:
                continue


            iso = row[
                0
            ]


            abund = []


            for x in row[
                1:
            ]:

                try:

                    abund.append(
                        float(
                            x
                        )
                    )

                except Exception:

                    abund.append(
                        0.0
                    )


            values[
                iso
            ] = abund


    return (
        samples,
        values
    )


# ============================================================
# 7. FPM NORMALIZATION
# ============================================================

def calculate_fpm(
    samples,
    matrix
):

    totals = [
        0.0
        for _ in samples
    ]


    for vals in matrix.values():

        for i, x in enumerate(
            vals
        ):

            totals[
                i
            ] += x


    fpm = {}


    for iso, vals in matrix.items():

        out = []


        for i, x in enumerate(
            vals
        ):

            total = totals[
                i
            ]


            if total > 0:

                out.append(
                    x
                    / total
                    * 1_000_000.0
                )

            else:

                out.append(
                    0.0
                )


        fpm[
            iso
        ] = out


    return fpm

def read_sqanti_target(
    path
):

    rows = []


    with open(
        path
    ) as fh:

        reader = csv.DictReader(
            fh,
            delimiter="\t"
        )


        for row in reader:

            gene = row.get(
                "associated_gene",
                ""
            )


            if gene != TARGET_GENE:

                continue


            # If filter_result exists, require retained Isoform.
            if (
                "filter_result" in row
                and row[
                    "filter_result"
                ] not in (
                    "",
                    "Isoform"
                )
            ):

                continue


            rows.append(
                row
            )


    return rows


# ============================================================
# 9. READ TARGET EXON STRUCTURES
# ============================================================

def read_target_gtf(
    path,
    target_isoforms
):

    exons = defaultdict(
        list
    )


    with open(
        path
    ) as fh:

        for line in fh:

            if (
                not line.strip()
                or line.startswith(
                    "#"
                )
            ):

                continue


            parts = line.rstrip().split(
                "\t"
            )


            if len(
                parts
            ) < 9:

                continue


            if parts[
                2
            ] != "exon":

                continue


            attr = parts[
                8
            ]


            m = re.search(
                r'transcript_id "([^"]+)"',
                attr
            )


            if not m:
                continue


            iso = m.group(
                1
            )


            if iso not in target_isoforms:
                continue


            exons[
                iso
            ].append(
                {

                    "chrom":
                        parts[
                            0
                        ],

                    "start":
                        int(
                            parts[
                                3
                            ]
                        ),

                    "end":
                        int(
                            parts[
                                4
                            ]
                        ),

                    "strand":
                        parts[
                            6
                        ]
                }
            )


    return exons

def build_sample_metadata(

    metadata = []


    for sample in samples:

                "treatment":
                    treatment

    return metadata


def group_means(
    isoform_values,
    metadata
):

    groups = defaultdict(
        list
    )


    for i, m in enumerate(
        metadata
    ):

        if (
            m[
                "genotype"
            ] is None
            or m[
                "treatment"
            ] is None
        ):

            continue


        key = (
            m[
                "genotype"
            ],
            m[
                "treatment"
            ]
        )


        groups[
            key
        ].append(
            isoform_values[
                i
            ]
        )


    means = {}


    for key, vals in groups.items():

        means[
            key
        ] = (
            sum(
                vals
            )
            / len(
                vals
            )
        )


    return means

def read_fasta(path):

    records = {}

    current = None
    seq = []


    with open(
        path
    ) as fh:

        for line in fh:

            line = line.strip()

            if not line:
                continue


            if line.startswith(">"):

                if current is not None:

                    records[
                        current
                    ] = "".join(
                        seq
                    ).upper()


                current = (
                    line[1:]
                    .split()[0]
                )

                seq = []


            else:

                seq.append(
                    line
                )


    if current is not None:

        records[
            current
        ] = "".join(
            seq
        ).upper()


    return records

def recover_protein(
    orf_seq
):

    seq = (
        orf_seq
        .strip()
        .upper()
        .replace(
            "*",
            ""
        )
    )


    if seq in (
        "",
        "NA",
        "NAN",
        "NONE"
    ):

        return ""


    # Protein sequence if characters occur outside nucleotide
    # alphabet.
    nucleotide_chars = set(
        "ACGTUN"
    )


    observed = set(
        seq
    )


    if not observed.issubset(
        nucleotide_chars
    ):

        # SQANTI ORF_seq is already amino acid.
        return seq


    # --------------------------------------------------------
    # If ORF_seq happens to be nucleotide in a catalog,
    # translate it.
    # --------------------------------------------------------

    codon_table = {

        "TTT":"F","TTC":"F","TTA":"L","TTG":"L",
        "TCT":"S","TCC":"S","TCA":"S","TCG":"S",
        "TAT":"Y","TAC":"Y","TAA":"*","TAG":"*",
        "TGT":"C","TGC":"C","TGA":"*","TGG":"W",

        "CTT":"L","CTC":"L","CTA":"L","CTG":"L",
        "CCT":"P","CCC":"P","CCA":"P","CCG":"P",
        "CAT":"H","CAC":"H","CAA":"Q","CAG":"Q",
        "CGT":"R","CGC":"R","CGA":"R","CGG":"R",

        "ATT":"I","ATC":"I","ATA":"I","ATG":"M",
        "ACT":"T","ACC":"T","ACA":"T","ACG":"T",
        "AAT":"N","AAC":"N","AAA":"K","AAG":"K",
        "AGT":"S","AGC":"S","AGA":"R","AGG":"R",

        "GTT":"V","GTC":"V","GTA":"V","GTG":"V",
        "GCT":"A","GCC":"A","GCA":"A","GCG":"A",
        "GAT":"D","GAC":"D","GAA":"E","GAG":"E",
        "GGT":"G","GGC":"G","GGA":"G","GGG":"G"
    }


    protein = []


    seq = seq.replace(
        "U",
        "T"
    )


    for i in range(
        0,
        len(seq) - 2,
        3
    ):

        protein.append(
            codon_table.get(
                seq[
                    i:i + 3
                ],
                "X"
            )
        )


    return "".join(
        protein
    ).rstrip(
        "*"
    )

#!/usr/bin/env python3

from pathlib import Path
import csv
from collections import defaultdict


# ============================================================
# UNBIASED SAG101 g309600 TRANSCRIPT / PROTEIN ANALYSIS
#
# Reference groups:
#
#   Vrip_X_Vrup:
#       101-14
#       3309
#       SW
#
#   Vrip_X_Vber:
#       SO4
#       5BB
#
#   Vrip_X_Vace:
#       1616
#
#   Vripa:
#       Ripa
#
#
# IMPORTANT:
#
# This script does NOT use P3 or P16 labels.
#
# It asks:
#
#   1. Which g309600 transcripts are expressed?
#   2. Which exact predicted proteins do they encode?
#   3. Which protein sequences recur across catalogs?
#   4. How abundant is each transcript/protein group?
#   5. How does protein-group usage change after infestation?
#   6. Which genomic transcript cluster produces each protein?
#
#
# Protein grouping:
#
#   Isoforms with IDENTICAL amino-acid sequences are grouped
#   into the same global protein group, even if their PB IDs
#   differ between Iso-Seq catalogs.
#
# ============================================================

def read_csv(path):

    with open(
        path,
        newline=""
    ) as fh:

        return list(
            csv.DictReader(
                fh
            )
        )


def write_csv(
    path,
    rows,
    fieldnames=None
):

    if not rows:

        print(
            f"WARNING: no rows to write: {path}"
        )

        return


    if fieldnames is None:

        fieldnames = []

        for row in rows:

            for key in row.keys():

                if key not in fieldnames:

                    fieldnames.append(
                        key
                    )


    with open(
        path,
        "w",
        newline=""
    ) as fh:

        writer = csv.DictWriter(
            fh,
            fieldnames=fieldnames,
            extrasaction="ignore"
        )

        writer.writeheader()

        writer.writerows(
            rows
        )


def to_float(x):

    if x in (
        None,
        "",
        "NA",
        "NaN",
        "nan"
    ):

        return 0.0

    try:

        return float(
            x
        )

    except Exception:

        return 0.0


def to_int(x):

    try:

        return int(
            float(
                x
            )
        )

    except Exception:

        return 0


# ============================================================
# 4. TRANSLATION
#
# SQANTI ORF_seq may already contain amino-acid sequence.
#
# If it contains characters outside the nucleotide alphabet,
# treat it as protein directly.
#
# Only translate when it actually looks like DNA/RNA.
# ============================================================

CODON_TABLE = {

    "TTT":"F","TTC":"F","TTA":"L","TTG":"L",
    "TCT":"S","TCC":"S","TCA":"S","TCG":"S",
    "TAT":"Y","TAC":"Y","TAA":"*","TAG":"*",
    "TGT":"C","TGC":"C","TGA":"*","TGG":"W",

    "CTT":"L","CTC":"L","CTA":"L","CTG":"L",
    "CCT":"P","CCC":"P","CCA":"P","CCG":"P",
    "CAT":"H","CAC":"H","CAA":"Q","CAG":"Q",
    "CGT":"R","CGC":"R","CGA":"R","CGG":"R",

    "ATT":"I","ATC":"I","ATA":"I","ATG":"M",
    "ACT":"T","ACC":"T","ACA":"T","ACG":"T",
    "AAT":"N","AAC":"N","AAA":"K","AAG":"K",
    "AGT":"S","AGC":"S","AGA":"R","AGG":"R",

    "GTT":"V","GTC":"V","GTA":"V","GTG":"V",
    "GCT":"A","GCC":"A","GCA":"A","GCG":"A",
    "GAT":"D","GAC":"D","GAA":"E","GAG":"E",
    "GGT":"G","GGC":"G","GGA":"G","GGG":"G"
}


def recover_protein(
    orf_seq
):

    if orf_seq is None:

        return ""


    seq = (
        orf_seq
        .strip()
        .upper()
        .replace(
            " ",
            ""
        )
    )


    if seq in (
        "",
        "NA",
        "NAN",
        "NONE"
    ):

        return ""


    # Remove only terminal stop symbols.
    seq = seq.rstrip(
        "*"
    )


    nucleotide_alphabet = set(
        "ACGTUN"
    )


    # --------------------------------------------------------
    # Already amino-acid sequence
    # --------------------------------------------------------

    if not set(
        seq
    ).issubset(
        nucleotide_alphabet
    ):

        return seq


    # --------------------------------------------------------
    # Otherwise translate nucleotide sequence
    # --------------------------------------------------------

    seq = seq.replace(
        "U",
        "T"
    )


    protein = []


    for i in range(
        0,
        len(seq) - 2,
        3
    ):

        codon = seq[
            i:i + 3
        ]


        protein.append(
            CODON_TABLE.get(
                codon,
                "X"
            )
        )


    return "".join(
        protein
    ).rstrip(
        "*"
    )


# ============================================================
# 5. LOAD ALL SQANTI PROTEIN SEQUENCES
#
# First pass:
#
#   reference
#   PB transcript
#   exact amino-acid sequence
#
# No abundance information is used yet.
# ============================================================

protein_by_reference_isoform = {}

classification_rows = {}


for reference in DATASETS:

    classification = (
        BASE
        / "sqanti_results"
        / f"{reference}_sqanti"
        / "filter_results"
        / f"{reference}_filtered_RulesFilter_result_classification.txt"
    )


    if not classification.exists():

        raise FileNotFoundError(
            f"Missing SQANTI classification:\n"
            f"{classification}"
        )


    rows_for_reference = {}


    with open(
        classification
    ) as fh:

        reader = csv.DictReader(
            fh,
            delimiter="\t"
        )


        for row in reader:

            if row.get(
                "associated_gene"
            ) != TARGET_GENE:

                continue


            # If filter_result exists, only retain
            # SQANTI-filtered Isoforms.
            if (
                "filter_result" in row
                and row.get(
                    "filter_result"
                ) not in (
                    "",
                    "Isoform"
                )
            ):

                continue


            iso = row.get(
                "isoform",
                ""
            )


            if not iso:

                continue


            protein = recover_protein(
                row.get(
                    "ORF_seq",
                    ""
                )
            )


            rows_for_reference[
                iso
            ] = row


            if protein:

                protein_by_reference_isoform[
                    (
                        reference,
                        iso
                    )
                ] = protein


    classification_rows[
        reference
    ] = rows_for_reference


# ============================================================
# 6. READ EXPRESSION / FRACTION / STRUCTURE OUTPUTS
#
# These were produced by:
#
# analyze_SAG101_g309600_across_Vrip_backgrounds.py
#
# We deliberately IGNORE its old predicted-protein-group file,
# because that file was based on the incorrect ORF translation.
# ============================================================

expression_tables = {}
fraction_tables = {}
structure_tables = {}


for reference in DATASETS:

    ref_dir = (
        INPUT_ROOT
        / reference
    )


    expression_file = (
        ref_dir
        / "g309600_isoform_expression_FPM.csv"
    )

    fraction_file = (
        ref_dir
        / "g309600_isoform_fraction.csv"
    )

    structure_file = (
        ref_dir
        / "g309600_transcript_structure.csv"
    )


    for f in [
        expression_file,
        fraction_file,
        structure_file
    ]:

        if not f.exists():

            raise FileNotFoundError(
                f"Missing required file:\n{f}"
            )


    expression_tables[
        reference
    ] = read_csv(
        expression_file
    )

    fraction_tables[
        reference
    ] = read_csv(
        fraction_file
    )

    structure_tables[
        reference
    ] = read_csv(
        structure_file
    )


# ============================================================
# 7. BUILD LOOKUPS
# ============================================================

expression_lookup = {}
fraction_lookup = {}
structure_lookup = {}


for reference in DATASETS:

    expression_lookup[
        reference
    ] = {

        row[
            "Isoform_ID"
        ]:
            row

        for row in expression_tables[
            reference
        ]
    }


    fraction_lookup[
        reference
    ] = {

        row[
            "Isoform_ID"
        ]:
            row

        for row in fraction_tables[
            reference
        ]
    }


    tmp = defaultdict(
        list
    )


    for row in structure_tables[
        reference
    ]:

        tmp[
            row[
                "Isoform_ID"
            ]
        ].append(
            row
        )


    structure_lookup[
        reference
    ] = tmp


# ============================================================
# 8. DETERMINE WHICH ISOFORMS ARE EXPRESSED
# ============================================================

expressed_keys = set()

display_keys = set()


for reference in DATASETS:

    for iso, row in expression_lookup[
        reference
    ].items():

        n_nonzero = to_int(
            row.get(
                "N_samples_nonzero"
            )
        )

        n_gt_01 = to_int(
            row.get(
                "N_samples_FPM_gt_0.1"
            )
        )


        if n_nonzero > 0:

            expressed_keys.add(
                (
                    reference,
                    iso
                )
            )


        if n_gt_01 >= 2:

            display_keys.add(
                (
                    reference,
                    iso
                )
            )


# ============================================================
# 9. GLOBAL EXACT PROTEIN GROUPING
#
# The SAME amino-acid sequence receives the SAME protein group
# across all four Iso-Seq catalogs.
# ============================================================

sequence_members = defaultdict(
    list
)


for key, protein in protein_by_reference_isoform.items():

    if key not in expressed_keys:

        continue


    sequence_members[
        protein
    ].append(
        key
    )


# ============================================================
# 10. CALCULATE GLOBAL ABUNDANCE FOR EACH EXACT PROTEIN
#
# This is used only to order the anonymous protein groups:
#
#   PG001 = most abundant exact protein sequence overall
#   PG002 = next most abundant
#   etc.
#
# No P3/P16 information is involved.
# ============================================================

sequence_total_mean_fpm = defaultdict(
    float
)


for protein_sequence, members in sequence_members.items():

    for reference, iso in members:

        expr = expression_lookup[
            reference
        ].get(
            iso,
            {}
        )


        for genotype in DATASETS[
            reference
        ]:

            for treatment in [
                "Control",
                "Infested"
            ]:

                col = (
                    f"{genotype}_{treatment}_mean_FPM"
                )


                sequence_total_mean_fpm[
                    protein_sequence
                ] += to_float(
                    expr.get(
                        col
                    )
                )


# Sort by:
#   1. global summed mean FPM descending
#   2. protein length descending
#   3. sequence alphabetically for deterministic tie-breaking

ordered_sequences = sorted(

    sequence_members.keys(),

    key=lambda seq: (
        -sequence_total_mean_fpm[
            seq
        ],
        -len(
            seq
        ),
        seq
    )
)


protein_group_for_sequence = {

    seq:
        f"PG{index:03d}"

    for index, seq in enumerate(
        ordered_sequences,
        start=1
    )
}


# ============================================================
# 11. BUILD ISOFORM-LEVEL MASTER TABLE
# ============================================================

all_isoform_rows = []


for reference, genotypes in DATASETS.items():

    for iso, expr in expression_lookup[
        reference
    ].items():

        key = (
            reference,
            iso
        )


        if key not in expressed_keys:

            continue


        frac = fraction_lookup[
            reference
        ].get(
            iso,
            {}
        )


        protein_sequence = (
            protein_by_reference_isoform.get(
                key,
                ""
            )
        )


        protein_group = ""


        if protein_sequence:

            protein_group = (
                protein_group_for_sequence[
                    protein_sequence
                ]
            )


        structure_rows = structure_lookup[
            reference
        ].get(
            iso,
            []
        )


        if structure_rows:

            tx_start = min(
                to_int(
                    x.get(
                        "Transcript_Start"
                    )
                )
                for x in structure_rows
            )

            tx_end = max(
                to_int(
                    x.get(
                        "Transcript_End"
                    )
                )
                for x in structure_rows
            )

            chrom = structure_rows[
                0
            ].get(
                "Chromosome",
                ""
            )

            strand = structure_rows[
                0
            ].get(
                "Strand",
                ""
            )

            exon_count = len(
                structure_rows
            )

        else:

            tx_start = ""
            tx_end = ""
            chrom = ""
            strand = ""
            exon_count = 0


        genomic_cluster = frac.get(
            "Genomic_Cluster",
            ""
        )


        row = {

            "Reference":
                reference,

            "Isoform_ID":
                iso,

            "Genomic_Cluster":
                genomic_cluster,

            "Chromosome":
                chrom,

            "Strand":
                strand,

            "Transcript_Start":
                tx_start,

            "Transcript_End":
                tx_end,

            "Exon_Count":
                exon_count,

            "Protein_Group":
                protein_group,

            "Protein_Length_aa":
                (
                    len(
                        protein_sequence
                    )
                    if protein_sequence
                    else ""
                ),

            "N_samples_nonzero":
                to_int(
                    expr.get(
                        "N_samples_nonzero"
                    )
                ),

            "N_samples_FPM_gt_0.1":
                to_int(
                    expr.get(
                        "N_samples_FPM_gt_0.1"
                    )
                ),

            "Pass_display_filter":
                key in display_keys
        }


        for genotype in genotypes:

            for treatment in [
                "Control",
                "Infested"
            ]:

                fpm_col = (
                    f"{genotype}_{treatment}_mean_FPM"
                )

                if_col = (
                    f"{genotype}_{treatment}_mean_IF"
                )


                row[
                    fpm_col
                ] = to_float(
                    expr.get(
                        fpm_col
                    )
                )


                row[
                    if_col
                ] = to_float(
                    frac.get(
                        if_col
                    )
                )


        for genotype in genotypes:

            control_fpm = to_float(
                row.get(
                    f"{genotype}_Control_mean_FPM"
                )
            )

            infested_fpm = to_float(
                row.get(
                    f"{genotype}_Infested_mean_FPM"
                )
            )

            control_if = to_float(
                row.get(
                    f"{genotype}_Control_mean_IF"
                )
            )

            infested_if = to_float(
                row.get(
                    f"{genotype}_Infested_mean_IF"
                )
            )


            row[
                f"{genotype}_Delta_FPM_Inf_minus_Control"
            ] = (
                infested_fpm
                - control_fpm
            )


            row[
                f"{genotype}_Delta_IF_Inf_minus_Control"
            ] = (
                infested_if
                - control_if
            )


        all_isoform_rows.append(
            row
        )


# ============================================================
# 12. GLOBAL PROTEIN-GROUP METADATA
# ============================================================

protein_metadata_rows = []


for protein_sequence in ordered_sequences:

    pg = protein_group_for_sequence[
        protein_sequence
    ]


    members = sequence_members[
        protein_sequence
    ]


    references = sorted(
        {
            x[
                0
            ]
            for x in members
        }
    )


    transcript_ids = [
        f"{reference}:{iso}"

        for reference, iso in sorted(
            members
        )
    ]


    clusters = set()


    for reference, iso in members:

        frac = fraction_lookup[
            reference
        ].get(
            iso,
            {}
        )


        cluster = frac.get(
            "Genomic_Cluster",
            ""
        )


        if cluster not in (
            "",
            None
        ):

            clusters.add(
                f"{reference}:cluster{cluster}"
            )


    protein_metadata_rows.append(
        {

            "Protein_Group":
                pg,

            "Protein_Length_aa":
                len(
                    protein_sequence
                ),

            "N_Expressed_Transcripts":
                len(
                    members
                ),

            "N_Reference_Catalogs":
                len(
                    references
                ),

            "Reference_Catalogs":
                ";".join(
                    references
                ),

            "Transcript_IDs":
                ";".join(
                    transcript_ids
                ),

            "Genomic_Clusters":
                ";".join(
                    sorted(
                        clusters
                    )
                ),

            "Global_Summed_Mean_FPM":
                sequence_total_mean_fpm[
                    protein_sequence
                ],

            "Protein_Sequence":
                protein_sequence
        }
    )


# ============================================================
# 13. PROTEIN EXPRESSION / USAGE BY GENOTYPE × TREATMENT
#
# One row per:
#
#   Reference
#   Genotype
#   Treatment
#   Protein_Group
#
# Absolute FPM:
#   summed across all transcripts encoding that exact protein.
#
# Relative fraction:
#   summed across all transcript isoform fractions encoding
#   that exact protein.
# ============================================================

protein_long_rows = []


for reference, genotypes in DATASETS.items():

    # Protein groups represented by expressed transcripts
    # in this particular catalog.
    groups_in_reference = sorted(
        {
            row[
                "Protein_Group"
            ]

            for row in all_isoform_rows

            if (
                row[
                    "Reference"
                ] == reference
                and row[
                    "Protein_Group"
                ] != ""
            )
        }
    )


    for genotype in genotypes:

        for treatment in [
            "Control",
            "Infested"
        ]:

            fpm_col = (
                f"{genotype}_{treatment}_mean_FPM"
            )

            if_col = (
                f"{genotype}_{treatment}_mean_IF"
            )


            for pg in groups_in_reference:

                member_rows = [

                    row

                    for row in all_isoform_rows

                    if (
                        row[
                            "Reference"
                        ] == reference
                        and row[
                            "Protein_Group"
                        ] == pg
                    )
                ]


                total_fpm = sum(
                    to_float(
                        row.get(
                            fpm_col
                        )
                    )
                    for row in member_rows
                )


                total_if = sum(
                    to_float(
                        row.get(
                            if_col
                        )
                    )
                    for row in member_rows
                )


                passing_members = [

                    row[
                        "Isoform_ID"
                    ]

                    for row in member_rows

                    if row[
                        "Pass_display_filter"
                    ]
                ]


                all_members = [

                    row[
                        "Isoform_ID"
                    ]

                    for row in member_rows
                ]


                protein_long_rows.append(
                    {

                        "Reference":
                            reference,

                        "Genotype":
                            genotype,

                        "Treatment":
                            treatment,

                        "Protein_Group":
                            pg,

                        "Protein_Length_aa":
                            next(
                                (
                                    row[
                                        "Protein_Length_aa"
                                    ]

                                    for row in member_rows

                                    if row[
                                        "Protein_Length_aa"
                                    ] != ""
                                ),
                                ""
                            ),

                        "Mean_FPM":
                            total_fpm,

                        "Mean_Protein_Fraction":
                            total_if,

                        "N_Expressed_Transcripts":
                            len(
                                member_rows
                            ),

                        "N_Display_Filter_Transcripts":
                            len(
                                passing_members
                            ),

                        "Expressed_Transcript_IDs":
                            ";".join(
                                sorted(
                                    all_members
                                )
                            ),

                        "Display_Filter_Transcript_IDs":
                            ";".join(
                                sorted(
                                    passing_members
                                )
                            )
                    }
                )


# ============================================================
# 14. INFESTED - CONTROL DELTAS
#
# One row per:
#
#   Reference
#   Genotype
#   Protein_Group
# ============================================================

protein_delta_rows = []


long_lookup = {}


for row in protein_long_rows:

    key = (
        row[
            "Reference"
        ],
        row[
            "Genotype"
        ],
        row[
            "Treatment"
        ],
        row[
            "Protein_Group"
        ]
    )


    long_lookup[
        key
    ] = row


for reference, genotypes in DATASETS.items():

    groups_in_reference = sorted(
        {
            row[
                "Protein_Group"
            ]

            for row in protein_long_rows

            if row[
                "Reference"
            ] == reference
        }
    )


    for genotype in genotypes:

        for pg in groups_in_reference:

            control = long_lookup.get(
                (
                    reference,
                    genotype,
                    "Control",
                    pg
                )
            )

            infested = long_lookup.get(
                (
                    reference,
                    genotype,
                    "Infested",
                    pg
                )
            )


            if control is None or infested is None:

                continue


            control_fpm = to_float(
                control[
                    "Mean_FPM"
                ]
            )

            infested_fpm = to_float(
                infested[
                    "Mean_FPM"
                ]
            )

            control_fraction = to_float(
                control[
                    "Mean_Protein_Fraction"
                ]
            )

            infested_fraction = to_float(
                infested[
                    "Mean_Protein_Fraction"
                ]
            )


            protein_delta_rows.append(
                {

                    "Reference":
                        reference,

                    "Genotype":
                        genotype,

                    "Protein_Group":
                        pg,

                    "Protein_Length_aa":
                        control[
                            "Protein_Length_aa"
                        ],

                    "Control_Mean_FPM":
                        control_fpm,

                    "Infested_Mean_FPM":
                        infested_fpm,

                    "Delta_FPM_Inf_minus_Control":
                        (
                            infested_fpm
                            - control_fpm
                        ),

                    "Control_Mean_Protein_Fraction":
                        control_fraction,

                    "Infested_Mean_Protein_Fraction":
                        infested_fraction,

                    "Delta_Protein_Fraction_Inf_minus_Control":
                        (
                            infested_fraction
                            - control_fraction
                        )
                }
            )


# ============================================================
# 15. DISPLAY-FILTERED PROTEIN GROUPS
#
# A protein group is considered display-supported in a catalog
# if at least one transcript encoding it has:
#
#       FPM > 0.1 in >= 2 samples
# ============================================================

display_supported = set()


for row in all_isoform_rows:

    if (
        row[
            "Pass_display_filter"
        ]
        and row[
            "Protein_Group"
        ] != ""
    ):

        display_supported.add(
            (
                row[
                    "Reference"
                ],
                row[
                    "Protein_Group"
                ]
            )
        )


display_protein_long_rows = [

    row

    for row in protein_long_rows

    if (
        row[
            "Reference"
        ],
        row[
            "Protein_Group"
        ]
    ) in display_supported
]


display_delta_rows = [

    row

    for row in protein_delta_rows

    if (
        row[
            "Reference"
        ],
        row[
            "Protein_Group"
        ]
    ) in display_supported
]


# ============================================================

write_csv(

write_csv(

            line = line.rstrip(

                    yield (
                        header,
                        "".join(
                            seq
                        )
                    )


                header = line[
                    1:
                ]


                seq = []


            else:

                seq.append(
                    line.strip()

                header,

        file
    )


    header_idx <- which(
        startsWith(
            x,
            ">"
        )
    )


    out <- vector(
        "list",
        length(
            header_idx
        )
    )


    for (
        i in seq_along(
            header_idx
        )
    ) {

        h <- header_idx[i]


        last_line <- if (
            i <
            length(
                header_idx
            )
        ) {

            header_idx[i + 1] - 1

        } else {

            length(
                x
            )
        }


        header <- sub(
            "^>",
            "",
            x[h]
        )


        id <- strsplit(
            header,
            "\\s+"
        )[[1]][1]


        seq <- if (
            last_line > h
        ) {

            paste0(
                x[
                    (h + 1):
                    last_line
                ],
                collapse = ""
            )

        } else {

            ""
        }


        seq <- gsub(
            "\\*",
            "",
            seq
        )


        out[[i]] <- data.frame(

            InterPro_ID = id,

            Sequence = seq,

            FASTA_Length_aa = nchar(
                seq
            ),

            stringsAsFactors = FALSE
        )
    }


    bind_rows(
        out
    )
}

                    Treatment = treatment,

                    .before = 1

                paste(
                    reference,
                    genotype,
                    treatment,
                    sep = "__"

make_plot <- function(
    reference
) {

    genotypes <- reference_genotypes[[
        reference
    ]]


    cat(
        "\n============================================================\n"
    )

    cat(
        reference,
        "\n"
    )

    cat(
        "============================================================\n"
    )


    dat_all <- master %>%

        filter(
            Reference == reference
        )


    if_cols <- unlist(
        lapply(
            genotypes,
            function(
                g
            ) {

                c(
                    paste0(
                        g,
                        "_Control_mean_IF"
                    ),
                    paste0(
                        g,
                        "_Infested_mean_IF"
                    )
                )
            }
        )
    )


    # ========================================================
    # DISPLAY FILTER
    # ========================================================

    dat <- dat_all %>%

        rowwise() %>%

        mutate(

            Max_Mean_IF = max(
                c_across(
                    all_of(
                        if_cols
                    )
                ),
                na.rm = TRUE
            )
        ) %>%

        ungroup() %>%

        filter(
            is.finite(
                Max_Mean_IF
            ),
            Max_Mean_IF >= MIN_PLOT_IF
        )


    cat(
        "Displayed isoforms:",
        nrow(
            dat
        ),
        "of",
        nrow(
            dat_all
        ),
        "\n"
    )


    if (
        nrow(
            dat
        ) == 0
    ) {

        return(
            NULL
        )
    }


    # ========================================================
    # ORDER TRANSCRIPTS
    #
    # Use genomic position so the two transcriptional units
    # remain visually apparent.
    # ========================================================

    dat <- dat %>%

        arrange(
            Transcript_Start,
            Transcript_End,
            Isoform_ID
        ) %>%

        mutate(
            y_pos = rev(
                seq_len(
                    n()
                )
            )
        )


    y_lookup <- dat %>%

        select(
            Isoform_ID,
            y_pos
        )


    # ========================================================
    # PANEL A — ISOFORM USAGE
    # ========================================================

    heat <- dat %>%

        select(
            Isoform_ID,
            y_pos,
            all_of(
                if_cols
            )
        ) %>%

        pivot_longer(

            cols = all_of(
                if_cols
            ),

            names_to = "Group",

            values_to = "Mean_IF"
        )


    group_levels <- if_cols


    group_labels <- unlist(
        lapply(
            genotypes,
            function(
                g
            ) {

                c(
                    paste0(
                        g,
                        "\nControl"
                    ),
                    paste0(
                        g,
                        "\nInfested"
                    )
                )
            }
        )
    )


    heat$Group <- factor(
        heat$Group,
        levels = group_levels,
        labels = group_labels
    )


    p_heat <- ggplot(
        heat,
        aes(
            x = Group,
            y = y_pos,
            fill = Mean_IF
        )
    ) +

        geom_tile(
            width = 0.86,
            height = 0.72
        ) +

        scale_fill_gradient(
            low = "white",
            high = "steelblue",
            limits = c(
                0,
                1
            ),
            name = "Mean isoform\nfraction"
        ) +

        scale_y_continuous(

            breaks = dat$y_pos,

            labels = dat$Isoform_ID,

            limits = c(
                0.5,
                nrow(
                    dat
                ) + 0.5
            )
        ) +

        labs(
            x = NULL,
            y = NULL,
            title = "Isoform usage"
        ) +

        theme_bw() +

        theme(

            panel.grid = element_blank(),

            axis.text.x = element_text(
                angle = 45,
                hjust = 1
            ),

            axis.text.y = element_text(
                size = 8
            ),

            legend.position = "bottom",

            plot.title = element_text(
                face = "bold",
                size = 10
            )
        )


    # ========================================================
    # PANEL B — TRANSCRIPT STRUCTURE
    # ========================================================

    EXON_FILE <- file.path(
        BASE,
        "SAG101_g309600_cross_genotype",
        reference,
        "g309600_transcript_structure.csv"
    )


    exon_dat <- read.csv(
        EXON_FILE,
        stringsAsFactors = FALSE,
        check.names = FALSE
    ) %>%

        filter(
            Isoform_ID %in%
                dat$Isoform_ID
        ) %>%

        inner_join(
            y_lookup,
            by = "Isoform_ID"
        )


    tx_ranges <- exon_dat %>%

        group_by(
            Isoform_ID,
            y_pos
        ) %>%

        summarise(

            xmin = min(
                Exon_Start
            ),

            xmax = max(
                Exon_End
            ),

            .groups = "drop"
        )


    p_rna <- ggplot() +

        geom_segment(

            data = tx_ranges,

            aes(
                x = xmin,
                xend = xmax,
                y = y_pos,
                yend = y_pos
            ),

            linewidth = 0.45
        ) +

        geom_rect(

            data = exon_dat,

            aes(
                xmin = Exon_Start,
                xmax = Exon_End,

                ymax = y_pos + 0.28
            )
        ) +

        scale_y_continuous(

            limits = c(
                0.5,
                nrow(
                    dat
                ) + 0.5
            )
        ) +

        labs(
            x = "Genomic coordinate",

# ============================================================
# 3. FASTA READER
# ============================================================

def read_fasta(
    path
):

    records = {}

    header = None
    sequence = []


    with open(
        path
    ) as fh:

        for line in fh:

            line = line.strip()


            if not line:

                continue


            if line.startswith(
                ">"
            ):

                if header is not None:

                    records[
                        header
                    ] = "".join(
                        sequence
                    )


                header = line[
                    1:
                ]


                sequence = []


            else:

                sequence.append(
                    line
                )


        if header is not None:

            records[
                header
            ] = "".join(
                sequence
            )


    return records

def get_unique_by_pattern(
    records,
    pattern
):

    matches = []


    for header, sequence in records.items():

        tokens = re.split(
            r"__|\|",
            header
        )


        if pattern in tokens:

            matches.append(
                (
                    header,
                    sequence
                )
            )


    if len(
        matches
    ) == 0:

        raise ValueError(
            f"No FASTA record matched exact label: "
            f"{pattern}"
        )


    unique_sequences = {}


    for header, sequence in matches:

        sequence = (
            sequence
            .replace(
                "*",
                ""
            )
            .upper()
        )


        unique_sequences.setdefault(
            sequence,
            []
        ).append(
            header
        )


    if len(
        unique_sequences
    ) != 1:

        raise ValueError(
            f"{pattern} matched "
            f"{len(unique_sequences)} unique sequences"
        )


    sequence = next(
        iter(
            unique_sequences
        )
    )


    return (
        sequence,
        unique_sequences[
            sequence
        ]
    )

def numeric_or_none(
    x
):

    if x is None:

        return None


    try:

        value = float(
            x
        )


        if math.isnan(
            value
        ):

            return None


        return value


    except:

        return None


# ============================================================
# 6. HELPER:
# LOCATE FULL CONFIDENCE FILE FOR A SUMMARY FILE
#
# AF3 normally gives:
#
#   *_summary_confidences.json
#   *_confidences.json
#
# in the same directory.
# ============================================================

def confidence_for_summary(
    summary_file
):

    name = summary_file.name


    if not name.endswith(
        "_summary_confidences.json"
    ):

        return None


    confidence_name = name.replace(
        "_summary_confidences.json",
        "_confidences.json"
    )


    confidence_file = (
        summary_file.parent
        / confidence_name
    )


    if confidence_file.exists():

        return confidence_file


    # --------------------------------------------------------
    # Fallback:
    # if exact paired name is absent, accept a single
    # confidence JSON in the same directory.
    # --------------------------------------------------------

    candidates = [

        x

        for x in summary_file.parent.glob(
            "*_confidences.json"
        )

        if "_summary_confidences.json"
        not in x.name
    ]


    if len(
        candidates
    ) == 1:

        return candidates[
            0
        ]


    return None

# 7. HELPER:
# CLASSIFY WHETHER RESULT IS A SEED/SAMPLE MODEL
# ============================================================

def seed_sample_from_path(
    path
):

    seed = None
    sample = None


    for part in path.parts:

        if part.startswith(
            "seed-"
        ) and "_sample-" in part:

            try:

                left, right = part.split(
                    "_sample-"
                )


                seed = int(
                    left.replace(
                        "seed-",
                        ""
                    )
                )


                sample = int(
                    right
                )


            except:

                pass


    return (
        seed,
        sample
    )

# 12. INTER-CHAIN PAE FUNCTION
# ============================================================

def analyze_interface(
    complex_name,
    model
):

    confidence_file = Path(
        model[
            "confidence_file"
        ]
    )


    with open(
        confidence_file
    ) as fh:

        confidence = json.load(
            fh
        )


    if "pae" not in confidence:

        raise ValueError(
            f"{complex_name}: confidence JSON lacks 'pae'\n"
            f"{confidence_file}"
        )


    if "token_chain_ids" not in confidence:

        raise ValueError(
            f"{complex_name}: confidence JSON lacks "
            f"'token_chain_ids'\n"
            f"{confidence_file}"
        )


    pae = np.asarray(
        confidence[
            "pae"
        ],
        dtype=float
    )


    chain_ids = list(
        confidence[
            "token_chain_ids"
        ]
    )


    unique_chains = []


    for chain in chain_ids:

        if chain not in unique_chains:

            unique_chains.append(
                chain
            )


    if len(
        unique_chains
    ) != 2:

        raise ValueError(
            f"{complex_name}: expected 2 chains, "
            f"found {unique_chains}"
        )


    chain_a = unique_chains[
        0
    ]


    chain_b = unique_chains[
        1
    ]


    idx_a = np.array(

        [
            i

            for i, chain in enumerate(
                chain_ids
            )

            if chain == chain_a
        ],

        dtype=int
    )


    idx_b = np.array(

        [
            i

            for i, chain in enumerate(
                chain_ids
            )

            if chain == chain_b
        ],

        dtype=int
    )


    if (
        len(
            idx_a
        ) == 0
        or
        len(
            idx_b
        ) == 0
    ):

        raise ValueError(
            f"{complex_name}: empty chain index."
        )


    # --------------------------------------------------------
    # Directional blocks:
    #
    #   A -> B
    #   B -> A
    #
    # As in the previous analysis, combine both directions.
    # --------------------------------------------------------

    ab = pae[
        np.ix_(
            idx_a,
            idx_b
        )
    ].ravel()


    ba = pae[
        np.ix_(
            idx_b,
            idx_a
        )
    ].ravel()


    combined = np.concatenate(
        [
            ab,
            ba
        ]
    )


    def fraction_below(
        cutoff
    ):

        return float(
            np.mean(
                combined < cutoff
            )
        )


    left, right = complex_name.split(
        "__"
    )


    eds1 = left.replace(
        "EDS1_",
        ""
    )


    sag101 = right.replace(
        "SAG101_",
        ""
    )


    return {

        "complex":
            complex_name,

        "EDS1":
            eds1,

        "SAG101":
            sag101,

        "best_seed":
            model[
                "seed"
            ],

        "best_sample":
            model[
                "sample"
            ],

        "ranking_score":
            model[
                "ranking_score"
            ],

        "iptm":
            model[
                "iptm"
            ],

        "ptm":
            model[
                "ptm"
            ],

        "chain_A":
            chain_a,

        "chain_B":
            chain_b,

        "n_tokens_A":
            len(
                idx_a
            ),

        "n_tokens_B":
            len(
                idx_b
            ),

        "n_interchain_values":
            len(
                combined
            ),

        "mean_interchain_PAE":
            float(
                np.mean(
                    combined
                )
            ),

        "median_interchain_PAE":
            float(
                np.median(
                    combined
                )
            ),

        "min_interchain_PAE":
            float(
                np.min(
                    combined
                )
            ),

        "fraction_interchain_PAE_lt5":
            fraction_below(
                5
            ),

        "fraction_interchain_PAE_lt10":
            fraction_below(
                10
            ),

        "fraction_interchain_PAE_lt15":
            fraction_below(
                15
            ),

        "summary_file":
            model[
                "summary_file"
            ],

        "confidence_file":
            model[
                "confidence_file"
            ]
    }

# 16. HEATMAP FUNCTION
# ============================================================

def plot_heatmap(
    matrix,
    title,
    xlabel,
    ylabel,
    cbar_label,
    outfile,
    decimals=2,
    vmin=None,
    vmax=None
):

    data = matrix.to_numpy(
        dtype=float
    )


    n_rows = len(
        matrix.index
    )


    n_cols = len(
        matrix.columns
    )


    width = max(
        4.8,
        1.4 * n_cols +
        2.5
    )


    height = max(
        4.2,
        0.9 * n_rows +
        2.0
    )


    fig, ax = plt.subplots(
        figsize=(
            width,
            height
        )
    )


    im = ax.imshow(

        data,

        aspect="auto",

        vmin=vmin,

        vmax=vmax
    )


    cbar = fig.colorbar(
        im,
        ax=ax,
        shrink=0.82
    )


    cbar.set_label(
        cbar_label
    )


    ax.set_xticks(
        np.arange(
            n_cols
        )
    )


    ax.set_xticklabels(
        matrix.columns
    )


    ax.set_yticks(
        np.arange(
            n_rows
        )
    )


    ax.set_yticklabels(
        matrix.index
    )


    ax.set_xlabel(
        xlabel
    )


    ax.set_ylabel(
        ylabel
    )


    ax.set_title(
        title
    )


    # --------------------------------------------------------
    # Cell values
    # --------------------------------------------------------

    for i in range(
        n_rows
    ):

        for j in range(
            n_cols
        ):

            value = data[
                i,
                j
            ]


            if np.isnan(
                value
            ):

                label = "NA"

            else:

                label = (
                    f"{value:.{decimals}f}"
                )


            ax.text(

                j,
                i,
                label,

                ha="center",
                va="center",

                fontsize=10
            )


    fig.tight_layout()


    fig.savefig(

        outfile.with_suffix(
            ".pdf"
        ),

        bbox_inches="tight"
    )


    fig.savefig(

        outfile.with_suffix(
            ".png"
        ),

        dpi=600,

        bbox_inches="tight"
    )


    plt.close(
        fig
    )

#

#

#

#       EDS1 <-> SAG101 inter-chain confidence

#   low PAE:
#       more confident relative positioning
#

#       less confident relative positioning
#
# ============================================================

def safe_float(
    value
):

    if value is None:

        return None


    try:

        x = float(
            value
        )


        if math.isnan(
            x
        ):

            return None


        return x


    except:

        return None


# ============================================================
# 5. FIND BEST MODEL FOR A COMPLEX
# ============================================================

def find_best_model(
    complex_name
):

    complex_dir = (
        OUTPUTS
        / complex_name
    )


    if not complex_dir.exists():

        print(
            "Missing complex directory:",
            complex_dir
        )

        return None


    summary_files = sorted(

        complex_dir.rglob(
            "*_summary_confidences.json"
        )
    )


    candidates = []


    for summary_file in summary_files:

        try:

            with open(
                summary_file
            ) as fh:

                summary = json.load(
                    fh
                )


        except Exception as error:

            print(
                "Could not read:",
                summary_file
            )

            print(
                error
            )

            continue


        # ----------------------------------------------------
        # Find corresponding full confidence file
        # ----------------------------------------------------

        confidence_name = (
            summary_file.name.replace(
                "_summary_confidences.json",
                "_confidences.json"
            )
        )


        confidence_file = (
            summary_file.parent
            / confidence_name
        )


        # ----------------------------------------------------
        # Fallback if exact paired filename not present
        # ----------------------------------------------------

        if not confidence_file.exists():

            fallback = [

                x

                for x in summary_file.parent.glob(
                    "*_confidences.json"
                )

                if "_summary_confidences.json"
                not in x.name
            ]


            if len(
                fallback
            ) == 1:

                confidence_file = fallback[
                    0
                ]


        if not confidence_file.exists():

            continue


        candidates.append({

            "summary_file":
                summary_file,

            "confidence_file":
                confidence_file,

            "ranking_score":
                safe_float(
                    summary.get(
                        "ranking_score"
                    )
                ),

            "iptm":
                safe_float(
                    summary.get(
                        "iptm"
                    )
                ),

            "ptm":
                safe_float(
                    summary.get(
                        "ptm"
                    )
                )
        })


    if len(
        candidates
    ) == 0:

        print(
            "No usable models found for:",
            complex_name
        )

        return None


    def score(
        row
    ):

        ranking = (
            row[
                "ranking_score"
            ]
            if row[
                "ranking_score"
            ] is not None
            else -np.inf
        )


        iptm = (
            row[
                "iptm"
            ]
            if row[
                "iptm"
            ] is not None
            else -np.inf
        )


        ptm = (
            row[
                "ptm"
            ]
            if row[
                "ptm"
            ] is not None
            else -np.inf
        )


        return (
            ranking,
            iptm,
            ptm
        )


    return max(
        candidates,
        key=score
    )
