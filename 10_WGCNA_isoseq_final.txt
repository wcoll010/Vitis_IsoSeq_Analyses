read_matrix <- function(geno) {

  f <- file.path(
    matrix_dir,
    paste0(
      geno,
      "_oarfish_counts.tsv"
    )
  )

  if (!file.exists(f)) {
    stop(
      "Missing matrix:\n",
      f
    )
  }

  x <- read.delim(
    f,
    header = TRUE,
    row.names = 1,
    check.names = FALSE
  )

  cat("\n============================================================\n")
  cat("GENOTYPE:", geno, "\n")
  cat("============================================================\n")

  cat(
    "Isoforms:",
    nrow(x),
    "\n"
  )

  cat(
    "Samples:",
    ncol(x),
    "\n"
  )

  cat("Sample names:\n")
  print(
    colnames(x)
  )

  return(x)
}


mat_list <- lapply(
  genotypes,
  read_matrix
)

names(mat_list) <- genotypes


# ============================================================
# 3. Check for duplicated sample names
# ============================================================

all_sample_names <- unlist(
  lapply(
    mat_list,
    colnames
  )
)

if (anyDuplicated(all_sample_names) > 0) {

  stop(
    "Duplicate sample names found across genotype matrices."
  )
}


cat("\n============================================================\n")
cat("SAMPLE SUMMARY\n")
cat("============================================================\n")

for (geno in genotypes) {

  cat(
    geno,

  )
}

)

# ============================================================

all_isoforms <- Reduce(
  union,
  lapply(
    mat_list,
    rownames
  )
)

cat(
  "\nUnique isoform IDs across all three matrices:",
  length(all_isoforms),
  "\n"
)


expand_matrix <- function(x, all_ids) {

  out <- matrix(
    0,
    nrow = length(all_ids),
    ncol = ncol(x),
    dimnames = list(
      all_ids,
      colnames(x)
    )
  )

  out[
    rownames(x),
  ] <- as.matrix(x)

  out
}


expanded <- lapply(
  mat_list,
  expand_matrix,
  all_ids = all_isoforms
)


isoform_counts <- do.call(
  cbind,
  expanded
)


cat(

  out <- matrix(
    0,
    nrow = length(ids),
    ncol = ncol(x),

      ids,

  gene_counts_integer

  row.names = colnames(gene_counts_integer),

  countData = gene_counts_integer,

  colData = sample_metadata,

  design = ~ 1

  dds,
  blind = TRUE
)

  vsd

  ncol(expr_normalized),

)

# ============================================================
# SIGNED WGCNA WORKFLOW
# Vrip_X_Vrup: 101-14 + 3309 + sw; gene-level network (isoforms collapsed); signed network, soft power = 16
# ============================================================

library(WGCNA)
library(pheatmap)
library(dplyr)
library(tidyr)
library(tibble)
library(writexl)

options(stringsAsFactors = FALSE)

allowWGCNAThreads(nThreads = 12)


# ============================================================
# 0. Paths
# ============================================================

BASE <- "/rhome/wcoll010/shared/pacbio/IsoSeq_analyses_2"

INPUT_FILE <- file.path(
  BASE,
  "WGCNA_Vrip_X_Vrup",
  "WGCNA_input_Vrip_X_Vrup.RData"
)

OUT_DIR <- file.path(
  BASE,
  "WGCNA_Vrip_X_Vrup",
  "signed_network"
)

dir.create(
  OUT_DIR,
  recursive = TRUE,
  showWarnings = FALSE
)


# ============================================================
# 1. Load prepared expression matrix
# ============================================================

load(INPUT_FILE)

cat("\nObjects loaded:\n")
print(ls())

cat("\ndatExpr dimensions:\n")
print(dim(datExpr))

input_mat <- datExpr

cat(
  "\nInput matrix:",
  nrow(input_mat),
  "samples x",
  ncol(input_mat),
  "genes\n"
)


# ============================================================
# 2. Check samples and genes
# ============================================================

gsg <- goodSamplesGenes(
  input_mat,
  verbose = 3
)

if (!gsg$allOK) {

  cat("\nRemoving bad samples/genes...\n")

  if (sum(!gsg$goodGenes) > 0) {
    cat("Bad genes:", sum(!gsg$goodGenes), "\n")
  }

  if (sum(!gsg$goodSamples) > 0) {
    cat("Bad samples:", sum(!gsg$goodSamples), "\n")
  }

  input_mat <- input_mat[
    gsg$goodSamples,
    gsg$goodGenes
  ]
}

cat(
  "\nFinal matrix:",
  nrow(input_mat),
  "samples x",
  ncol(input_mat),
  "genes\n"
)


# ============================================================
# 3. Signed soft-threshold analysis
# ============================================================

powers <- c(
  1:10,
  seq(12, 30, by = 2)
)

sft_signed <- pickSoftThreshold(
  input_mat,
  powerVector = powers,
  networkType = "signed",
  verbose = 5
)


# ============================================================
# 4. Save soft-threshold results
# ============================================================

write.csv(
  sft_signed$fitIndices,
  file.path(
    OUT_DIR,
    "signed_soft_threshold_fit_indices.csv"
  ),
  row.names = FALSE
)

save(
  sft_signed,
  powers,
  file = file.path(
    OUT_DIR,
    "signed_soft_threshold_results.RData"
  )
)


# ============================================================
# 5. Plot soft-threshold results
# ============================================================

signed_R2 <-
  -sign(sft_signed$fitIndices[, 3]) *
  sft_signed$fitIndices[, 2]

pdf(
  file.path(
    OUT_DIR,
    "signed_soft_threshold_plots.pdf"
  ),
  width = 10,
  height = 5
)

par(mfrow = c(1, 2))

plot(
  sft_signed$fitIndices[, 1],
  signed_R2,
  xlab = "Soft Threshold (power)",
  ylab = "Scale Free Topology Model Fit, signed R^2",
  main = "Scale independence",
  type = "n"
)

text(
  sft_signed$fitIndices[, 1],
  signed_R2,
  labels = sft_signed$fitIndices[, 1],
  cex = 0.9,
  col = "red"
)

  h = 0.85,

)

)

text(
  sft_signed$fitIndices[, 1],

# ============================================================
# 6. Set selected signed soft power
# ============================================================

softPower <- 16

cat(
  "\nUsing signed soft-threshold power:",
  softPower,
  "\n"
)


# ============================================================
# 7. Build signed adjacency matrix
# ============================================================

adjacency_mat <- adjacency(
  input_mat,
  power = softPower,
  type = "signed"
)

cat("\nAdjacency matrix complete.\n")


# ============================================================
# 8. Convert adjacency to signed TOM
# ============================================================

TOM <- TOMsimilarity(
  adjacency_mat,
  TOMType = "signed"
)

rm(adjacency_mat)
gc()


# ============================================================
# 9. TOM dissimilarity
# ============================================================

dissTOM <- 1 - TOM

rm(TOM)
gc()


# ============================================================
# 10. Hierarchical clustering
# ============================================================

geneTree <- hclust(
  as.dist(dissTOM),
  method = "average"
)


# ============================================================
# 11. Plot gene dendrogram
# ============================================================

pdf(
  file.path(
    OUT_DIR,
    "signed_power16_gene_dendrogram.pdf"

  width = 12,
  height = 9
)

plot(
  geneTree,
  xlab = "",
  sub = "",
  main = "Gene clustering on signed TOM-based dissimilarity",
  labels = FALSE,
  hang = 0.04
)

dev.off()

# 12. Dynamic tree cut
# ============================================================

minModuleSize <- 20

  dendro = geneTree,
  distM = dissTOM,

)

    table(dynamicMods),
    decreasing = TRUE
  )
)

# ============================================================

)

  sort(
    table(dynamicColors),
    decreasing = TRUE
  )
)

cat(

)

  "\n"

# ============================================================
# 6. Type III contrasts
# ============================================================

old_contrasts <- options("contrasts")

options(
  contrasts = c(
    "contr.sum",
    "contr.poly"
  )
)


# ============================================================
# 7. Regression:
#
# ME ~ Genotype * Treatment
#
# Tests:
#   Genotype
#   Treatment
#   Genotype x Treatment
# ============================================================

results_list <- vector(
  "list",
  length(module_cols)
)

names(results_list) <- module_cols


for (mod in module_cols) {

  cat("Testing:", mod, "\n")

  form <- reformulate(
    "Genotype * Treatment",
    response = mod
  )

  fit <- lm(
    form,
    data = dat
  )

  fit_anova <- car::Anova(
    fit,
    type = 3
  )

  fit_summary <- summary(fit)

  results_list[[mod]] <- data.frame(

    Module_Eigengene = mod,

    Module = sub(
      "^ME",
      "",
      mod
    ),

    P_Genotype =
      fit_anova["Genotype", "Pr(>F)"],

    P_Treatment =
      fit_anova["Treatment", "Pr(>F)"],

    P_GxT =
      fit_anova["Genotype:Treatment", "Pr(>F)"],

    R_squared =
      fit_summary$r.squared,

    Adjusted_R_squared =
      fit_summary$adj.r.squared,

    stringsAsFactors = FALSE
  )
}


# Restore contrasts
options(
  contrasts = old_contrasts$contrasts
)


# ============================================================
# 8. Combine modules + FDR correction
# ============================================================

  results_list
) %>%

  mutate(

    FDR_Genotype = p.adjust(
      P_Genotype,
      method = "BH"
    ),

    FDR_Treatment = p.adjust(
      P_Treatment,
      method = "BH"
    ),

    FDR_GxT = p.adjust(
      P_GxT,
      method = "BH"
    ),

    Sig_Genotype =
      FDR_Genotype < 0.05,

    Sig_Treatment =
      FDR_Treatment < 0.05,

    Sig_GxT =
      FDR_GxT < 0.05

  ) %>%

  arrange(
    FDR_GxT,
    FDR_Treatment,
    FDR_Genotype
  )

# ============================================================
# 9. Function:
# Remove exact duplicate contributing-gene sets FOR PLOTTING
#
# Example:
#
# Pathway A -> gene1/gene4/gene8/gene10
# Pathway B -> gene1/gene4/gene8/gene10
#
# Only the pathway with the smaller adjusted P-value is
# retained in the plot.
#
# IMPORTANT:
# This does NOT alter the full enrichment-result CSV.
# ============================================================

remove_duplicate_gene_sets <- function(result_df) {

  if (nrow(result_df) == 0) {
    return(result_df)
  }

  result_df %>%

    rowwise() %>%

    mutate(

      GeneSet = paste(
        sort(
          unique(
            strsplit(
              geneID,
              "/",
              fixed = TRUE
            )[[1]]
          )
        ),
        collapse = "/"
      )

    ) %>%

    ungroup() %>%

    arrange(
      p.adjust,
      pvalue
    ) %>%

    distinct(
      GeneSet,
      .keep_all = TRUE
    ) %>%

    dplyr::select(
      -GeneSet
    )
}

    "KEGG-annotated background genes:",
    length(kegg_universe),
    "\n"
  )

  cat(
    "KEGG-annotated module genes:",
    length(kegg_targets),
    "\n"
  )


  if (length(kegg_targets) >= 5) {

    k_res <- enricher(

      gene = kegg_targets,

      universe = kegg_universe,

      TERM2GENE = KEGG_links %>%
        dplyr::select(
          ID,
          Gene
        ),

      TERM2NAME = KEGG_names,

      pvalueCutoff = 0.05,

      pAdjustMethod = "BH",

      minGSSize = 5
    )


    if (
      !is.null(k_res) &&
      nrow(as.data.frame(k_res)) > 0
    ) {

      # ------------------------------------------------------
      # Same final significance/count criterion as previous
      # workflow
      # ------------------------------------------------------

      k_res@result <- k_res@result %>%

        filter(
          Count >= 5
        )


      if (nrow(k_res@result) > 0) {

        # ====================================================
        # FULL significant results
        #
        # No gene-set redundancy removed here.
        # ====================================================

        kegg_full <- as.data.frame(
          k_res
        )


        write.csv(
          kegg_full,
          file.path(
            OUT_DIR,
            paste0(
              "Enrichment_KEGG_ME",
              mod,
              "_results.csv"
            )
          ),
          row.names = FALSE
        )


        # ====================================================
        # Contributing genes for all significant pathways
        # ====================================================

        kegg_gene_list <- kegg_full %>%

          dplyr::select(
            ID,
            Description,
            GeneRatio,
            BgRatio,
            pvalue,
            p.adjust,
            Count,
            geneID
          ) %>%

          separate_rows(
            geneID,
            sep = "/"
          ) %>%

          dplyr::rename(
            Gene_ID = geneID
          )


        write.csv(
          kegg_gene_list,
          file.path(
            OUT_DIR,
            paste0(
              "Enrichment_KEGG_ME",
              mod,
              "_gene_list.csv"
            )
          ),
          row.names = FALSE
        )


        # ====================================================
        # Remove EXACT duplicate gene sets for PLOT only
        # ====================================================

        kegg_plot_df <- remove_duplicate_gene_sets(
          kegg_full
        )


        cat(
          "Significant KEGG pathways:",
          nrow(kegg_full),
          "\n"
        )

        cat(
          "Unique contributing gene sets for plot:",
          nrow(kegg_plot_df),
          "\n"
        )


        # Save the exact set of pathways displayed
        write.csv(
          kegg_plot_df,
          file.path(
            OUT_DIR,
            paste0(
              "Enrichment_KEGG_ME",
              mod,
              "_PLOTTED_results.csv"
            )
          ),
          row.names = FALSE
        )


        # Put reduced result back into enrichment object
        k_plot <- k_res

        k_plot@result <- kegg_plot_df


        # ====================================================
        # KEGG dotplot
        # ====================================================

        pdf(
          file.path(
            OUT_DIR,
            paste0(
              "Enrichment_KEGG_ME",
              mod,
              "_dotplot.pdf"
            )
          ),
          width = 11,
          height = ifelse(
            nrow(kegg_plot_df) < 10,
            7,
            9
          )
        )

          dotplot(
            k_plot,
            showCategory = 20,
            label_format = 55
          ) +

              paste(

                mod
              )
            ) +

            theme_bw()
        )

      }

    } else {

      cat(
        "[-] No significant KEGG pathways.\n"
      )
    }

  } else {

    cat(
      "[-] Too few KEGG-annotated module genes.\n"
    )
  }


  # ==========================================================
  # ==========================================================
  # 12. GO ENRICHMENT
  # ==========================================================
  # ==========================================================

  for (ont in c(
    "BP",
    "MF",
    "CC"
  )) {

    cat(
      "\n[+] Running GO",
      ont,
      "...\n"
    )


    ont_links <- GO_links %>%

      filter(
        ONTOLOGY == ont
      ) %>%

      dplyr::select(
        ID,
        Gene
      ) %>%

      distinct()


    ont_names <- go_master_map %>%

      filter(
        ONTOLOGY == ont
      ) %>%

      dplyr::select(
        ID,
        Description
      ) %>%

      distinct()


    if (nrow(ont_links) == 0) {
      next
    }


    # --------------------------------------------------------
    # GO-specific annotated background
    # --------------------------------------------------------

    go_universe <- unique(
      ont_links$Gene
    )

    go_targets <- intersect(
      target_genes,
      go_universe
    )


    cat(
      "GO",
      ont,
      "background genes:",
      length(go_universe),
      "\n"
    )

    cat(
      "GO",
      ont,
      "module genes:",
      length(go_targets),
      "\n"
    )


    if (length(go_targets) < 5) {
      next
    }


    # ========================================================
    # GO enrichment
    # ========================================================

    go_res <- enricher(

      gene = go_targets,

      universe = go_universe,

      TERM2GENE = ont_links,

      TERM2NAME = ont_names,

      pvalueCutoff = 0.05,

      pAdjustMethod = "BH",

      minGSSize = 5
    )


    if (
      is.null(go_res) ||
      nrow(as.data.frame(go_res)) == 0
    ) {

      cat(
        "[-] No significant GO",
        ont,
        "terms.\n"
      )

      next
    }


    # Same Count threshold as previous workflow
    go_res@result <- go_res@result %>%

      filter(
        Count >= 5
      )


    if (nrow(go_res@result) == 0) {
      next
    }


    # ========================================================
    # FULL significant GO results
    # ========================================================

    go_full <- as.data.frame(
      go_res
    )


    write.csv(
      go_full,
      file.path(
        OUT_DIR,
        paste0(
          "Enrichment_GO_",
          ont,
          "_ME",
          mod,
          "_results.csv"
        )
      ),
      row.names = FALSE
    )


    # ========================================================
    # Contributing genes
    # ========================================================

    go_gene_list <- go_full %>%

      dplyr::select(
        ID,
        Description,
        GeneRatio,
        BgRatio,
        pvalue,
        p.adjust,
        Count,
        geneID
      ) %>%

      separate_rows(
        geneID,
        sep = "/"
      ) %>%

      dplyr::rename(
        Gene_ID = geneID
      )


    write.csv(
      go_gene_list,
      file.path(
        OUT_DIR,
        paste0(
          "Enrichment_GO_",
          ont,
          "_ME",
          mod,
          "_gene_list.csv"
        )
      ),
      row.names = FALSE
    )


    # ========================================================
    # Remove exact duplicate contributing-gene sets for
    # plotting only.
    #
    # This is NOT semantic GO simplification.
    #
    # Related-but-distinct GO terms are still retained.
    # ========================================================

    go_plot_df <- remove_duplicate_gene_sets(
      go_full
    )


    cat(
      "Significant GO",
      ont,
      "terms:",
      nrow(go_full),
      "\n"
    )

    cat(
      "Unique contributing gene sets for plot:",
      nrow(go_plot_df),
      "\n"
    )


    write.csv(
      go_plot_df,
      file.path(
        OUT_DIR,
        paste0(
          "Enrichment_GO_",
          ont,
          "_ME",
          mod,
          "_PLOTTED_results.csv"
        )
      ),
      row.names = FALSE
    )


    go_plot <- go_res
    go_plot@result <- go_plot_df


    # ========================================================
    # GO dotplot
    # ========================================================

    pdf(
      file.path(
        OUT_DIR,
        paste0(
          "Enrichment_GO_",
          ont,
          "_ME",
          mod,
          "_dotplot.pdf"
        )
      ),
      width = 11,
      height = ifelse(
        nrow(go_plot_df) < 10,
        7,
        9
      )
    )


    print(

      dotplot(
        go_plot,
        showCategory = 20,
        label_format = 50
      ) +

        ggtitle(
          paste(
            "GO",
            ont,
            "Enrichment:",
            mod
          )
        ) +

        theme_bw()
    )


    dev.off()
  }
}

  "midnightblue"
)

  cat("Plotting:", mod, "\n")

  moduleGenes <-
    hub_master_table$Module_Color == mod


  # Absolute module membership
  mm_values <- abs(
    hub_master_table[

      mm_col
    ]
  )


  # Absolute gene significance for infestation
  gs_values <- abs(
    hub_master_table[
      moduleGenes,
      "GS.Treatment"
    ]
  )


  # ==========================================================
  # Correlation
  # ==========================================================

  cor_val <- cor(
    mm_values,
    gs_values,
    use = "p"
  )

  pval <- corPvalueStudent(
    cor_val,
    sum(
      complete.cases(
        mm_values,
        gs_values
      )
    )
  )


  # ==========================================================
  # Linear regression
  # ==========================================================

  fit <- lm(
    gs_values ~ mm_values
  )


  # ==========================================================
  # Plot
  # ==========================================================

  pdf(
    file.path(
      HUB_DIR,
      paste0(
        "MM_vs_GS_Treatment_",
        mod,
        ".pdf"
      )
    ),
    width = 7,
    height = 7
  )


  plot(
    mm_values,
    gs_values,

    pch = 1,
    col = "blue",

    xlab = paste(
      "Module Membership in",
      mod,
      "module (absolute)"
    ),

    ylab =
      "Gene Significance for Infestation (absolute)",

    main = paste(
      "Module Membership vs. Gene Significance:",
      mod
    )
  )


  # Add fitted linear regression line
  abline(
    fit,
    col = "red",
    lwd = 2
  )


  # Add correlation statistics once
  legend(
    "topleft",

    legend = paste0(
      "r = ",
      signif(cor_val, 2),
      "\np = ",
      format.pval(
        pval,
        digits = 2,
        eps = 1e-300
      )
    ),

    bty = "n"
  )


  dev.off()
