### First, filter GTF files (generated by SQANTI3) so that only isoforms present in the isoform count data are retained \
\ this prevents the IsoformSwitchAnalyzeR program from crashing during DIU analysis due to Jaccard dissimilarity (genes present in the GTF but not in the isoform count file)



# Define genotypes and base directory
GENOTYPES=("101-14" "1103" "1616" "3309" "5BB" "99R" "cab33" "ripa" "SO4" "sw")
BASE_DIR="/rhome/wcoll010/bigdata_groenlab/iso_seq"

for GENO in "${GENOTYPES[@]}"; do
    echo "Starting filtering for genotype: $GENO"
    
    # Define paths
    COUNTS_CSV="${BASE_DIR}/final_abundance_matrices/${GENO}_counts_for_DE.csv"
    IDS_TXT="${BASE_DIR}/${GENO}_final_ids.txt"
    RAW_GTF="${BASE_DIR}/sqanti_results/${GENO}_sqanti/filter_results/${GENO}_filtered.filtered.gtf"
    CLEAN_GTF="${BASE_DIR}/sqanti_results/${GENO}_sqanti/filter_results/${GENO}_filtered.gtf"
    
    # 1. Extract valid isoform IDs from the header/first column of the CSV matrix
    awk -F',' 'NR>1 {gsub(/"/, "", $1); print $1}' "$COUNTS_CSV" > "$IDS_TXT"
    
    # 2. Filter the GTF to keep only those isoform IDs
    grep -Fwf "$IDS_TXT" "$RAW_GTF" > "$CLEAN_GTF"
    
    echo "Finished filtering $GENO. Clean GTF saved to: $CLEAN_GTF"
    echo "---------------------------------------------------"
done





### Perform differential isoform usage (DIU) calculations with IsoformSwitchAnalyzeR

# These lines allow the script to take 'genotype' as an input from the shell
args <- commandArgs(trailingOnly = TRUE)
if (length(args) == 0) {
    stop("Error: Please supply a genotype argument (e.g., Rscript genotype_DIU_array.R 1103)")
}
genotype <- args[1] 

my_lib <- "/rhome/wcoll010/bigdata_groenlab/R_libs"
.libPaths(c(my_lib, .libPaths()))
library(IsoformSwitchAnalyzeR)

# 1. Load data
counts_path <- paste0("/rhome/wcoll010/bigdata_groenlab/iso_seq/final_abundance_matrices/", genotype, "_counts_for_DE.csv")
class_path  <- paste0("/rhome/wcoll010/bigdata_groenlab/iso_seq/sqanti_results/", genotype, "_sqanti/filter_results/", genotype, "_filtered_RulesFilter_result_classification.txt")
# FIX: Points to the clean GTF generated in step 1
gtf_path    <- paste0("/rhome/wcoll010/bigdata_groenlab/iso_seq/sqanti_results/", genotype, "_sqanti/filter_results/", genotype, "_filtered.gtf")

counts_data <- read.csv(counts_path, header=TRUE, row.names=1, check.names=FALSE)
class_data  <- read.table(class_path, header=TRUE, sep="\t", quote=NULL)

# --- STEP 2: Filter and Map ---
iso_map <- class_data[class_data$filter_result == "Isoform", c("isoform", "associated_gene")]
colnames(iso_map) <- c("isoform_id", "gene_id")

# Synchronize counts with the filtered map
counts_data <- counts_data[rownames(counts_data) %in% iso_map$isoform_id, ]

# --- STEP 3: Design Matrix ---
design_matrix <- data.frame(
    sampleID = colnames(counts_data),
    condition = ifelse(grepl("con", colnames(counts_data)), "Control", "Infested")
)
design_matrix$condition <- factor(design_matrix$condition, levels = c("Control", "Infested"))

# --- STEP 4: Import & ID Swap ---
aSwitchList <- importRdata(
    isoformCountMatrix = counts_data,
    designMatrix = design_matrix,
    isoformExonAnnoation = gtf_path,
    showProgress = FALSE
)

aSwitchList$isoformFeatures$gene_id <- iso_map$gene_id[match(
    aSwitchList$isoformFeatures$isoform_id, 
    iso_map$isoform_id
)]

# --- STEP 5: Statistical Testing ---
aSwitchListFiltered <- preFilter(aSwitchList, geneExpressionCutoff = 1, removeSingleIsoformGenes = TRUE)
aSwitchListAnalyzed <- isoformSwitchTestDEXSeq(aSwitchListFiltered)

# --- STEP 6: Output ---
saveRDS(aSwitchListAnalyzed, file = paste0(genotype, "_DIU_results_final.rds"))

# Save summary stats and top switches
summary_stats <- extractSwitchSummary(aSwitchListAnalyzed)
write.csv(summary_stats, file = paste0(genotype, "_DIU_summary_stats.csv"), row.names = FALSE)

sig_switches <- extractTopSwitches(
    aSwitchListAnalyzed, 
    alpha = 0.05, 
    dIFcutoff = 0.1,
    n = Inf
)
write.csv(sig_switches, file = paste0(genotype, "_significant_isoform_switches.csv"), row.names = FALSE)
